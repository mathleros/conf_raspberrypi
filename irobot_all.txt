"""Settings for the local LLM client."""

from __future__ import annotations

from collections.abc import Mapping
from dataclasses import dataclass
import os


@dataclass(frozen=True)
class LLMSettings:
    """Local llama-server settings, overridable through environment variables."""

    model: str = "qwen2.5-1.5b"
    connect_timeout_seconds: float = 1.0
    read_timeout_seconds: float = 10.0
    max_tokens_plan: int = 120


def _env_positive(environ: Mapping[str, str], key: str, default: float, cast=float):
    try:
        value = cast(environ[key])
    except (KeyError, TypeError, ValueError):
        return default
    return value if value > 0 else default


def load_llm_settings(environ: Mapping[str, str] | None = None) -> LLMSettings:
    """Read LLM_* variables; missing or invalid values use defaults."""
    env = os.environ if environ is None else environ
    defaults = LLMSettings()
    return LLMSettings(
        model=(env.get("LLM_MODEL") or "").strip() or defaults.model,
        connect_timeout_seconds=_env_positive(env, "LLM_CONNECT_TIMEOUT_SECONDS", defaults.connect_timeout_seconds),
        read_timeout_seconds=_env_positive(env, "LLM_READ_TIMEOUT_SECONDS", defaults.read_timeout_seconds),
        max_tokens_plan=_env_positive(env, "LLM_MAX_TOKENS_PLAN", defaults.max_tokens_plan, int),
    )

@dataclass(frozen=True)
class MarkerIdentity:
    family: str
    dictionary: str
    identifier: str

    def __post_init__(self) -> None:
        if any(not isinstance(value, str) or not value.strip() or value != value.strip()
               for value in (self.family, self.dictionary, self.identifier)):
            raise ValueError("Marker identity fields must be non-empty canonical strings")

    @classmethod
    def from_mapping(cls, value: object) -> "MarkerIdentity | None":
        if value is None:
            return None
        if not isinstance(value, Mapping):
            raise ValueError("human_marker must be a table")
        expected = {"family", "dictionary", "identifier"}
        if set(value) != expected:
            raise ValueError("human_marker requires exactly family, dictionary, and identifier")
        try:
            return cls(
                family=value["family"],
                dictionary=value["dictionary"],
                identifier=value["identifier"],
            )
        except TypeError as exc:
            raise ValueError("human_marker identity values must be strings") from exc

def _table(raw: Mapping[str, object], name: str, allowed: set[str]) -> Mapping[str, object]:
    value = raw.get(name, {})
    if not isinstance(value, Mapping):
        raise ValueError(f"{name} must be a TOML table")
    unknown = set(value) - allowed
    if unknown:
        raise ValueError(f"Unknown key(s) in [{name}]: {', '.join(sorted(unknown))}")
    return value


def _float(table: Mapping[str, object], key: str, default: float, error_name: str) -> float:
    value = table.get(key, default)
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise ValueError(f"{error_name} must be a finite number")
    try:
        finite = math.isfinite(value)
    except OverflowError:
        finite = False
    if not finite:
        raise ValueError(f"{error_name} must be a finite number")
    return float(value)


def _int(table: Mapping[str, object], key: str, default: int, error_name: str) -> int:
    value = table.get(key, default)
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError(f"{error_name} must be an integer")
    return value


def _bool(table: Mapping[str, object], key: str, default: bool, error_name: str) -> bool:
    value = table.get(key, default)
    if not isinstance(value, bool):
        raise ValueError(f"{error_name} must be a boolean")
    return value


def _string(table: Mapping[str, object], key: str, default: str, error_name: str) -> str:
    value = table.get(key, default)
    if not isinstance(value, str) or not value.strip():
        raise ValueError(f"{error_name} must be a non-empty string")
    return value


def _size_px(table: Mapping[str, object], key: str, default: tuple[int, int], error_name: str) -> tuple[int, int]:
    value = table.get(key, default)
    if (not isinstance(value, (list, tuple)) or len(value) != 2
            or any(isinstance(item, bool) or not isinstance(item, int) or item <= 0 for item in value)):
        raise ValueError(f"{error_name} must contain two positive integers")
    return value[0], value[1]

def _load_optional_sections(raw: Mapping[str, object], current: RuntimeSettings) -> RuntimeSettings:
    human_follow = _table(raw, "human_follow", {"max_age_seconds", "announcement_timeout_seconds"})
    return replace(
        current,
        human_marker_identity=MarkerIdentity.from_mapping(raw.get("human_marker")),
        human_max_age_seconds=_float(
            human_follow, "max_age_seconds", current.human_max_age_seconds, "human_follow.max_age_seconds",
        ),
        human_announcement_timeout_seconds=_float(
            human_follow, "announcement_timeout_seconds", current.human_announcement_timeout_seconds,
            "human_follow.announcement_timeout_seconds",
        ),
    )


def load_runtime_settings() -> RuntimeSettings:
    path = _config_file()
    if not path.exists():
        return RuntimeSettings()

    raw = tomllib.loads(path.read_text(encoding="utf-8"))
    if not isinstance(raw, Mapping):
        raise ValueError("Configuration TOML must contain a table")
    allowed_sections = {"runtime", "limits", "camera", "human_follow", "human_marker"}
    unknown = set(raw) - allowed_sections
    if unknown:
        raise ValueError(f"Unknown top-level key(s): {', '.join(sorted(unknown))}")

    settings = RuntimeSettings()
    settings = _load_runtime_section(raw, settings)
    settings = _load_limits_section(raw, settings)
    settings = _load_camera_section(raw, settings, path)
    return _load_optional_sections(raw, settings)

from irobot.domain.intent import Intent


class IntentValidator:
    """Validate intent names and normalized tracking targets."""

    ALLOWED_INTENTS = {
        "follow_person",
        "follow_red_ball",
        "avoid_obstacle",
        "search_and_approach_object",
        "follow_object",
        "contour_obstacle",
        "spin_360",
        "move_square",
        "return_to_start",
        "move_forward",
        "move_backward",
        "turn_left",
        "turn_right",
        "stop",
    }

    def validate(self, intent: Intent) -> bool:
        if intent.name not in self.ALLOWED_INTENTS:
            return False
        if intent.name in {"follow_red_ball", "follow_object"}:
            targets = {None, "red_ball"} if intent.name == "follow_red_ball" else {None, "red", "yellow"}
            target_object = intent.parameters.get("target_object")
            return (
                intent.target in targets
                and (
                    target_object is None
                    or isinstance(target_object, str)
                    and target_object in {"red", "yellow", "red_ball", "yellow_ball", "marker"}
                )
            )
        return True

from __future__ import annotations

from irobot.domain.skills.base_skill import Skill, SkillContext, SkillResult


class SkillExecutor:
    """Exécute une skill avec validation et gestion de résultat."""

    def __init__(self, context: SkillContext):
        self.context = context

    def run(self, skill: Skill) -> SkillResult:
        validate_result = getattr(self.context.robot, "is_protected_result_current", None)
        if callable(validate_result):
            self.context.state["last_skill_result"] = None
        if callable(validate_result) and not validate_result():
            return SkillResult(
                status="failed", message="Contexte invalidé par la supervision.",
                data={"protection_fault": True}, confidence=0.0,
            )
        if not skill.validate(self.context):
            return SkillResult(
                status="failed",
                message=f"La skill '{skill.name}' n'est pas exécutable dans ce contexte.",
                data={"skill": skill.name},
                confidence=0.0,
                next_action=None,
            )

        if callable(validate_result) and not validate_result():
            return SkillResult(
                status="failed", message="Contexte invalidé par la supervision.",
                data={"protection_fault": True}, confidence=0.0,
            )
        result = skill.execute(self.context)
        if callable(validate_result) and not validate_result():
            return SkillResult(
                status="failed", message="Résultat invalidé par la supervision.",
                data={"protection_fault": True}, confidence=0.0,
            )
        self.context.state["last_skill_result"] = result
        return result

### FILE: src/irobot/domain/orchestrator/mission_orchestrator.py
"""Executes a planned mission with deadlines, retries, and state transitions."""

from __future__ import annotations

import math
import time
from collections.abc import Callable, Sequence
from typing import Any

from irobot.config.settings import RuntimeSettings, load_runtime_settings
from irobot.domain.intent import Intent
from irobot.domain.orchestrator.mission_state_machine import MissionStateMachine
from irobot.domain.orchestrator.planner import SkillPlanStep
from irobot.domain.skills.behaviors.approach_target import ApproachTargetSkill
from irobot.domain.skills.behaviors.target_tracking import FOLLOW_INTENTS


class MissionOrchestrator:
    """Runs compiled skill steps without depending on runtime or hardware interfaces."""

    def __init__(
        self,
        settings: RuntimeSettings | None = None,
        state_machine: MissionStateMachine | None = None,
        time_fn: Callable[[], float] | None = None,
    ) -> None:
        self.settings = settings or load_runtime_settings()
        self.state_machine = state_machine or MissionStateMachine()
        self.time_fn = time_fn or time.monotonic

    @staticmethod
    def _to_float(value: Any, default: float | None = None) -> float | None:
        try:
            number = float(value)
        except (TypeError, ValueError):
            return default
        return number if math.isfinite(number) else default

    @staticmethod
    def _step_has_failure(step_result: dict[str, Any]) -> bool:
        return any(
            getattr(result, "status", None) == "failed"
            for result in step_result.get("results") or []
        )

    @staticmethod
    def _step_in_progress(step_result: dict[str, Any]) -> bool:
        return any(
            getattr(result, "status", None) == "in_progress"
            for result in step_result.get("results") or []
        )

    def _safety_stop_reason(
        self,
        step_result: dict[str, Any],
        stop_requested: Callable[[], bool],
    ) -> str | None:
        if step_result.get("reason") == "obstacle prioritized":
            return "obstacle"
        if stop_requested():
            return "emergency stop"
        for result in step_result.get("results") or []:
            data = getattr(result, "data", None) or {}
            if data.get("emergency_stop"):
                return "emergency stop"
            if data.get("obstacle_detected") is True:
                return "obstacle"
        return None

    def _effective_step_timeout(self, intent: Intent) -> float:
        base = float(self.settings.step_timeout_seconds)
        params = intent.parameters or {}
        if intent.name in FOLLOW_INTENTS and params.get("duration_s") is None:
            return float("inf")
        if intent.name == "search_and_approach_object":
            tracking_s = ApproachTargetSkill._to_float(
                params.get("tracking_timeout_s"), ApproachTargetSkill.DEFAULT_MAX_TRACKING_S
            )
            search_s = ApproachTargetSkill._to_float(
                params.get("search_timeout_s"), float(self.settings.search_timeout_seconds)
            )
            return max(
                base,
                max(0.1, tracking_s) + max(0.1, search_s) + float(self.settings.timeout_margin_seconds),
            )
        duration_s = self._to_float(params.get("duration_s"))
        if duration_s is None:
            return base
        search_s = self._to_float(params.get("search_timeout_s"))
        if search_s is None:
            search_s = float(self.settings.search_timeout_seconds)
        return max(base, duration_s + search_s + float(self.settings.timeout_margin_seconds))

    def _effective_mission_timeout(self, intents: Sequence[Intent]) -> float:
        total = sum(self._effective_step_timeout(intent) for intent in intents)
        return max(float(self.settings.mission_timeout_seconds), total)

    def _execute_planned_step_with_retry(
        self,
        step: SkillPlanStep,
        execute_step: Callable[[SkillPlanStep], dict[str, Any]],
        stop_requested: Callable[[], bool],
        on_event: Callable[[dict[str, Any]], None] | None = None,
    ) -> dict[str, Any]:
        intent = step.intent
        following = intent.name in FOLLOW_INTENTS
        max_retries = 0 if following else max(0, int(self.settings.max_step_retries))
        step_timeout = self._effective_step_timeout(intent)
        attempt = 0
        last_result: dict[str, Any] | None = None
        while attempt <= max_retries:
            if stop_requested():
                return {"intent": intent, "status": "aborted", "reason": "emergency stop"}
            result = self._execute_attempt(
                step, execute_step, stop_requested, on_event, attempt + 1, step_timeout,
            )
            if result.get("status") in {"aborted", "stopped"}:
                return result
            if following and (self._step_has_failure(result) or self._step_in_progress(result)):
                result.update(status="failed", reason="tracking did not complete")
                return result
            if (
                result.get("status") == "accepted"
                and not self._step_has_failure(result)
                and not self._step_in_progress(result)
            ):
                return result

            last_result = result
            attempt += 1

        if last_result is None:
            return {"intent": intent, "status": "failed", "reason": "unknown execution error"}
        last_result["status"] = "incomplete"
        last_result["reason"] = last_result.get("reason") or "max retries reached"
        return last_result

    def _execute_attempt(
        self, step: SkillPlanStep,
        execute_step: Callable[[SkillPlanStep], dict[str, Any]],
        stop_requested: Callable[[], bool],
        on_event: Callable[[dict[str, Any]], None] | None,
        attempt: int, step_timeout: float,
    ) -> dict[str, Any]:
        started = self.time_fn()
        result = execute_step(step)
        duration = self.time_fn() - started
        result.update(attempt=attempt, step_duration_seconds=round(duration, 4))
        if on_event is not None:
            on_event({
                "event": "step_result", "intent": step.intent, "result": result,
                "duration_seconds": duration, "attempt": attempt,
            })
        reason = self._safety_stop_reason(result, stop_requested)
        if reason is not None:
            result.update(status="aborted", reason=reason)
            if on_event is not None:
                on_event({
                    "event": "emergency_stop" if reason == "emergency stop" else "obstacle_stop",
                    "reason": reason, "intent": step.intent,
                })
        elif duration > step_timeout:
            result.update(status="aborted", reason="step timeout")
        return result

    def execute(
        self,
        planned_steps: Sequence[SkillPlanStep],
        execute_step: Callable[[SkillPlanStep], dict[str, Any]],
        *,
        stop_requested: Callable[[], bool],
        on_event: Callable[[dict[str, Any]], None] | None = None,
    ) -> tuple[list[dict[str, Any]], str]:
        if not planned_steps:
            raise ValueError("A mission requires at least one planned step")
        intents = [step.intent for step in planned_steps]
        self.state_machine.start(
            intents[0].name,
            intents[0].target,
            metadata={"steps_total": len(intents)},
        )
        self.state_machine.begin_execution()
        try:
            steps, status = self._run_steps(planned_steps, execute_step, stop_requested, on_event)
        except Exception as exc:
            self.state_machine.fail(f"{type(exc).__name__}: {exc}")
            raise
        if self.state_machine.state.phase == "executing":
            self.state_machine.succeed()
        return steps, status

    def _run_steps(
        self, planned_steps: Sequence[SkillPlanStep],
        execute_step: Callable[[SkillPlanStep], dict[str, Any]],
        stop_requested: Callable[[], bool],
        on_event: Callable[[dict[str, Any]], None] | None,
    ) -> tuple[list[dict[str, Any]], str]:
        execution_started_at = self.time_fn()
        intents = [step.intent for step in planned_steps]
        mission_timeout = self._effective_mission_timeout(intents)
        steps: list[dict[str, Any]] = []
        global_status = "accepted"

        for planned_step in planned_steps:
            intent = planned_step.intent
            self.state_machine.state.metadata["step_index"] = len(steps)
            if self.time_fn() - execution_started_at > mission_timeout:
                global_status = "aborted"
                step_result = {"intent": intent, "status": "aborted", "reason": "mission timeout"}
                steps.append(step_result)
                self.state_machine.abort("mission timeout")
                if on_event is not None:
                    on_event({"event": "mission_timeout", "intent": intent})
                break

            step_result = self._execute_planned_step_with_retry(
                planned_step,
                execute_step,
                stop_requested,
                on_event,
            )
            if step_result.get("status") == "stopped":
                global_status = "stopped"
                steps.append(step_result)
                self.state_machine.abort("stop")
                break
            failed = step_result.get("status") in {"rejected", "failed", "incomplete", "aborted"}
            if failed:
                global_status = "aborted" if step_result.get("status") == "aborted" else "incomplete"
                if global_status == "aborted":
                    self.state_machine.abort(step_result.get("reason", "aborted"))
                else:
                    self.state_machine.fail(step_result.get("reason", "step failed"))
            steps.append(step_result)
            if failed:
                break

        return steps, global_status

from __future__ import annotations

"""Machine d'états d'une mission robotique.

Cette machine d'états constitue le premier niveau de supervision logicielle.
Elle formalise la progression d'une mission sous forme de phases explicites :
idle -> planning -> executing -> success / failed / aborted.
"""

from dataclasses import dataclass, field
from typing import Any


@dataclass
class MissionState:
    """État courant d'une mission."""

    phase: str = "idle"
    intent: str | None = None
    target: str | None = None
    last_error: str | None = None
    metadata: dict[str, Any] = field(default_factory=dict)


class MissionStateMachine:
    """Machine d'états pour orchestrer une mission robotique."""

    VALID_PHASES = {"idle", "planning", "executing", "success", "failed", "aborted"}

    def __init__(self):
        self.state = MissionState()

    def start(self, intent: str, target: str | None = None, metadata: dict[str, Any] | None = None) -> None:
        if not intent or not intent.strip():
            self.state.phase = "failed"
            self.state.last_error = f"Intent inconnue : {intent}"
            return

        self.state.phase = "planning"
        self.state.intent = intent
        self.state.target = target
        self.state.metadata = metadata or {}

    def begin_execution(self) -> None:
        self.state.phase = "executing"

    def succeed(self) -> None:
        self.state.phase = "success"

    def fail(self, error: str) -> None:
        self.state.phase = "failed"
        self.state.last_error = error
        self.state.metadata["failed_at"] = self.state.metadata.get("step_index")

    def abort(self, reason: str) -> None:
        self.state.phase = "aborted"
        self.state.last_error = reason
        self.state.metadata["aborted_at"] = self.state.metadata.get("step_index")

    def reset(self) -> None:
        self.state = MissionState()

    def as_dict(self) -> dict[str, Any]:
        return {
            "phase": self.state.phase,
            "intent": self.state.intent,
            "target": self.state.target,
            "last_error": self.state.last_error,
            "metadata": self.state.metadata,
        }

from collections.abc import Sequence
from dataclasses import dataclass

from irobot.config.settings import load_runtime_settings

@dataclass(frozen=True)
class SkillPlanStep:
    """One validated intent paired with the skills that implement it."""

    intent: Intent
    skills: tuple[Skill, ...]


class SkillPlanner:
    """Expands intents into executable steps and resolves their registered skills."""

    INTENT_TO_SKILLS: dict[str, list[str]] = {
        "follow_person": ["follow_object"],
        "follow_red_ball": ["follow_object"],
        "search_and_approach_object": [
            "detect_object",
            "approach_target",
            "detect_obstacle",
            "stop_motion",
        ],
        "follow_object": ["follow_object"],
        "move_forward": ["move_forward"],
        "move_backward": ["move_backward"],
        "turn_left": ["turn_left"],
        "turn_right": ["turn_right"],
        "move_circle": [
            "turn_left",
            "move_forward",
            "stop_motion",
        ],
        "stop": ["stop_motion"],
        "avoid_obstacle": [
            "detect_obstacle",
            "stop_motion",
            "turn_left",
            "move_forward",
        ],
    }

    def __init__(self, skills: Sequence[Skill] | None = None, registry: SkillRegistry | None = None):
        self.registry = registry or SkillRegistry.default()
        if skills:
            self.registry.register_many(skills)
        self.skills = list(skills or [])

    def plan(self, intent: str) -> list[Skill]:
        names = self.INTENT_TO_SKILLS.get(intent, [])
        if not names:
            return []

        resolved = self.registry.get_many(names)
        if resolved:
            return resolved

        return self.skills

    def plan_with_target(self, intent: str, *, target: str | None = None) -> list[Skill]:
        plan = self.plan(intent)
        if target == "red_ball" and "detect_object" not in {skill.name for skill in plan}:
            detect_skill = self.registry.get("detect_object")
            if detect_skill is not None:
                plan.insert(0, detect_skill)
        return plan

    def expand_intents(self, intents: Sequence[Intent]) -> list[Intent]:
        """Lower compound intents to an ordered sequence of executable intents."""
        expanded = self._expand_execute_path(list(intents))
        expanded = self._expand_high_level_intents(expanded)
        return self._expand_return_to_start(expanded)

    def plan_intents(self, intents: Sequence[Intent]) -> list[SkillPlanStep]:
        """Compile executable intents into typed intent-to-skills plan steps."""
        return [
            SkillPlanStep(intent=intent, skills=tuple(self.plan(intent.name)))
            for intent in intents
        ]

    def _expand_return_to_start(self, intents: list[Intent]) -> list[Intent]:
        reverse_map = {
            "move_forward": "move_backward",
            "move_backward": "move_forward",
            "turn_left": "turn_right",
            "turn_right": "turn_left",
        }
        history: list[Intent] = []
        expanded: list[Intent] = []
        for intent in intents:
            if intent.name == "return_to_start":
                for previous in reversed(history):
                    reverse_name = reverse_map.get(previous.name)
                    if reverse_name is None:
                        continue
                    expanded.append(
                        Intent(
                            name=reverse_name,
                            target=previous.target,
                            parameters=dict(previous.parameters),
                            confidence=previous.confidence,
                            raw_text=previous.raw_text,
                        )
                    )
                continue
            expanded.append(intent)
            if intent.name in reverse_map:
                history.append(intent)
        return expanded

    def _expand_execute_path(self, intents: list[Intent]) -> list[Intent]:
        action_synonyms = {
            "forward": "move_forward",
            "move": "move_forward",
            "backward": "move_backward",
            "back": "move_backward",
            "left": "turn_left",
            "right": "turn_right",
        }
        allowed_actions = {"move_forward", "move_backward", "turn_left", "turn_right", "stop"}
        expanded: list[Intent] = []
        for intent in intents:
            if intent.name != "execute_path":
                expanded.append(intent)
                continue
            steps = intent.parameters.get("steps")
            if not isinstance(steps, list) or not steps:
                expanded.append(
                    Intent(name="unknown", raw_text=intent.raw_text, parameters={"reason": "empty execute_path"})
                )
                continue
            for raw_step in steps[:MAX_PLAN_STEPS]:
                expanded.append(self._path_step(intent, raw_step, action_synonyms, allowed_actions))
        return expanded

    @staticmethod
    def _path_step(
        intent: Intent,
        raw_step: Any,
        synonyms: dict[str, str],
        allowed: set[str],
    ) -> Intent:
        if not isinstance(raw_step, dict):
            return Intent(
                name="unknown",
                raw_text=intent.raw_text,
                parameters={"reason": "invalid step format"},
            )
        action_raw = str(
            raw_step.get("action") or raw_step.get("intent") or raw_step.get("name") or ""
        ).strip().lower()
        action = synonyms.get(action_raw, action_raw)
        if action not in allowed:
            return Intent(
                name="unknown",
                raw_text=intent.raw_text,
                parameters={"reason": "invalid action", "action": action_raw},
            )
        step_params = raw_step.get("parameters")
        merged_params = dict(step_params) if isinstance(step_params, dict) else {}
        for key in ("distance", "unit", "angle", "duration"):
            if key in raw_step and key not in merged_params:
                merged_params[key] = raw_step[key]
        return Intent(
            name=action,
            target=intent.target,
            parameters=merged_params,
            confidence=float(raw_step.get("confidence") or intent.confidence or 0.0),
            raw_text=intent.raw_text,
        )

    def _expand_high_level_intents(self, intents: list[Intent]) -> list[Intent]:
        return [step for intent in intents for step in self._high_level_steps(intent)]

    @staticmethod
    def _derived_intent(intent: Intent, name: str, parameters: dict[str, Any]) -> Intent:
        return Intent(
            name=name,
            target=intent.target,
            parameters=parameters,
            confidence=intent.confidence,
            raw_text=intent.raw_text,
        )

    @staticmethod
    def _high_level_number(value: Any, default: float) -> float:
        try:
            return float(value)
        except (TypeError, ValueError):
            return default

    def _high_level_steps(self, intent: Intent) -> list[Intent]:
        params = intent.parameters
        if intent.name == "spin_360":
            return self._rotation_steps(intent, "turn_right")
        if intent.name in {"turn_right", "turn_left"} and bool(params.get("full_rotation")):
            return self._rotation_steps(intent, intent.name)
        if intent.name == "move_square":
            side = self._high_level_number(params.get("side", params.get("distance", 40.0)), 40.0)
            unit = str(params.get("unit", "cm"))
            direction = str(params.get("turn_direction", "right")).lower()
            turn = "turn_left" if direction == "left" else "turn_right"
            return [
                self._derived_intent(intent, name, values)
                for _ in range(4)
                for name, values in (
                    ("move_forward", {"distance": side, "unit": unit}),
                    (turn, {"angle": 90, "unit": "degrees"}),
                )
            ]
        if intent.name == "contour_obstacle":
            return self._contour_steps(intent)
        return [intent]

    def _rotation_steps(self, intent: Intent, direction: str) -> list[Intent]:
        return [
            self._derived_intent(intent, direction, {"angle": 90, "unit": "degrees"})
            for _ in range(4)
        ]

    def _contour_steps(self, intent: Intent) -> list[Intent]:
        params = intent.parameters
        clearance = self._high_level_number(params.get("clearance", params.get("distance", 20.0)), 20.0)
        unit = str(params.get("unit", "cm"))
        actions = (
            ("turn_right", {"angle": 45, "unit": "degrees"}),
            ("move_forward", {"distance": clearance, "unit": unit}),
            ("turn_left", {"angle": 90, "unit": "degrees"}),
            ("move_forward", {"distance": clearance, "unit": unit}),
            ("turn_right", {"angle": 45, "unit": "degrees"}),
        )
        return [self._derived_intent(intent, name, dict(values)) for name, values in actions]

from collections.abc import Iterable

from irobot.domain.skills import (

class SkillRegistry:
    """Registre central des skills exposées par le robot.

    L'objectif est de rendre le planificateur explicite et déterministe :
    le LLM ne décide pas directement des GPIO, il formule une intention,
    et le registre retourne la liste de skills associée à cette intention.
    """

    def __init__(self, skills: Iterable[Skill] | None = None):
        self._skills: dict[str, Skill] = {}
        if skills:
            for skill in skills:
                self.register(skill)

    def register(self, skill: Skill) -> Skill:
        self._skills[skill.name] = skill
        return skill

    def register_many(self, skills: Iterable[Skill]) -> None:
        for skill in skills:
            self.register(skill)

    def get(self, name: str) -> Skill | None:
        return self._skills.get(name)

    def get_many(self, names: Iterable[str]) -> list[Skill]:
        resolved: list[Skill] = []
        for name in names:
            skill = self.get(name)
            if skill is not None:
                resolved.append(skill)
        return resolved

    def names(self) -> list[str]:
        return sorted(self._skills)

    def __contains__(self, name: str) -> bool:
        return name in self._skills

    def __iter__(self):
        return iter(self._skills.values())

    @classmethod
    def default(cls) -> "SkillRegistry":
        registry = cls()
        registry.register_many(
            [
                MoveForwardSkill(),
                MoveBackwardSkill(),
                TurnLeftSkill(),
                TurnRightSkill(),
                StopMotionSkill(),
                DetectObjectSkill(),
                DetectObstacleSkill(),
                ApproachTargetSkill(),
                FollowObjectSkill(),
                MoveCircleSkill(),
            ]
        )
        return registry

from __future__ import annotations

import time
from collections.abc import Callable
from typing import Any

from irobot.config.settings import RuntimeSettings, load_runtime_settings
from irobot.domain.skills.base_skill import Skill, SkillContext, SkillResult
from irobot.domain.skills.behaviors.target_tracking import TargetTrackingMixin


class ApproachTargetSkill(TargetTrackingMixin, Skill):
    """Behavior composite : rejoint une cible détectée jusqu'à la distance demandée.

    Ce behavior est un exemple de logique de niveau supérieur : il combine
    perception et primitives de déplacement.
    """

    name = "approach_target"
    description = "Rejoint une cible visuelle jusqu'à sa distance de consigne"

    CONTROL_LOOP_PERIOD_S = 0.02
    DEFAULT_ALIGN_STEP_S = 0.08
    DEFAULT_FORWARD_STEP_S = 0.12
    DEFAULT_MAX_TRACKING_S = 8.0
    DEFAULT_MAX_TARGET_LOSS_S = 1.0
    DEFAULT_APPROACH_THRESHOLD_CM = 20.0
    DEFAULT_STAND_OFF_TOLERANCE_CM = 10.0
    MIN_FORWARD_SCALE = 0.25
    PHASE_SEARCH = "search"
    PHASE_TRACK = "track"

    def __init__(
        self,
        settings: RuntimeSettings | None = None,
        time_fn: Callable[[], float] = time.monotonic,
        sleep_fn: Callable[[float], None] = time.sleep,
    ) -> None:
        self._settings = settings
        self.time_fn = time_fn
        self.sleep_fn = sleep_fn

    @property
    def settings(self) -> RuntimeSettings:
        if self._settings is None:
            self._settings = load_runtime_settings()
        return self._settings

    def _resolve_stand_off_cm(self, params: dict[str, Any]) -> float:
        raw = params.get("approach_threshold_cm")
        if raw is None:
            raw = params.get("stand_off_cm")
        if raw is None:
            return self.DEFAULT_APPROACH_THRESHOLD_CM
        value = self._to_float(raw, self.DEFAULT_APPROACH_THRESHOLD_CM)
        low = float(self.settings.min_stand_off_cm)
        high = float(self.settings.max_stand_off_cm)
        return min(max(value, low), high)

    def _forward_step(self, base_step_s: float, distance_cm: float | None, stand_off_cm: float) -> float:
        # Impulsions raccourcies à l'approche pour ne pas dépasser le stand-off.
        if distance_cm is None or stand_off_cm <= 0.0 or distance_cm >= 2.0 * stand_off_cm:
            return base_step_s
        ratio = (distance_cm - stand_off_cm) / stand_off_cm
        return base_step_s * min(1.0, max(self.MIN_FORWARD_SCALE, ratio))

    def validate(self, context: SkillContext) -> bool:
        return context.robot is not None

    def execute(self, context: SkillContext) -> SkillResult:
        try:
            if not hasattr(context.robot, "forward"):
                return SkillResult(
                    status="failed",
                    message="Le robot ne fournit pas de méthode forward() pour rejoindre la cible.",
                    data={"skill": self.name},
                    confidence=0.0,
                    next_action=None,
                )
            params = dict(context.state.get("active_intent_parameters") or {})
            max_tracking_s = self._to_float(params.get("tracking_timeout_s"), self.DEFAULT_MAX_TRACKING_S)
            max_target_loss_s = self._to_float(params.get("max_target_loss_s"), self.DEFAULT_MAX_TARGET_LOSS_S)
            align_step_s = self._to_float(params.get("alignment_step_s"), self.DEFAULT_ALIGN_STEP_S)
            forward_step_s = self._to_float(params.get("forward_step_s"), self.DEFAULT_FORWARD_STEP_S)
            search_timeout_s = self._to_float(
                params.get("search_timeout_s"),
                float(self.settings.search_timeout_seconds),
            )
            stand_off_cm = self._resolve_stand_off_cm(params)
            tolerance_cm = max(
                0.0,
                self._to_float(params.get("stand_off_tolerance_cm"), self.DEFAULT_STAND_OFF_TOLERANCE_CM),
            )
            reached_threshold_cm = max(1.0, stand_off_cm + tolerance_cm)
            has_live_perception = callable(getattr(context.perception, "detect", None))

            if not has_live_perception:
                if self._check_obstacle_fail_safe(context):
                    self._stop_robot(context)
                    context.state["motion"] = "stop"
                    context.state["follow_target_state"] = {"phase": "blocked", "iterations": 1}
                    return SkillResult(
                        status="success",
                        message="Arrêt de sécurité : obstacle détecté pendant le suivi.",
                        data={"target_position": context.state.get("last_object_position"), "obstacle_detected": True},
                        confidence=0.99,
                        next_action="stop_motion",
                    )

                target_position, distance_cm = self._detect_target(context)
                if distance_cm is not None and distance_cm <= reached_threshold_cm:
                    self._stop_robot(context)
                    context.state["motion"] = "stop"
                    context.state["close_enough"] = True
                    context.state["follow_target_state"] = {
                        "phase": "target_reached",
                        "iterations": 1,
                        "distance_cm": distance_cm,
                    }
                    return SkillResult(
                        status="success",
                        message=f"Cible atteinte à {distance_cm:.1f} cm.",
                        data={"target_position": target_position, "distance_cm": distance_cm, "close_enough": True},
                        confidence=0.95,
                        next_action="stop_motion",
                    )

                if target_position is None:
                    self._pulse_motion(context, "right", align_step_s)
                    context.state["motion"] = "search_target"
                    context.state["follow_target_state"] = {"phase": "searching", "iterations": 1}
                    return SkillResult(
                        status="in_progress",
                        message="Cible non détectée : recherche active en rotation.",
                        data={"target_position": None, "obstacle_detected": False},
                        confidence=0.4,
                        next_action="detect_object",
                    )

                if target_position == "left":
                    self._pulse_motion(context, "left", align_step_s)
                    context.state["motion"] = "align_left"
                    context.state["follow_target_state"] = {
                        "phase": "aligning",
                        "direction": "left",
                        "iterations": 1,
                    }
                    return SkillResult(
                        status="in_progress",

                        data={"target_position": target_position, "obstacle_detected": False},
                        confidence=0.7,
                        next_action="detect_object",
                    )

                    self._pulse_motion(context, "right", align_step_s)
                    context.state["motion"] = "align_right"
                    context.state["follow_target_state"] = {
                        "phase": "aligning",
                        "direction": "right",
                        "iterations": 1,

                        data={"target_position": target_position, "obstacle_detected": False},

                        next_action="detect_object",
                    )

                context.state["motion"] = "approach_target"
                context.state["follow_target_state"] = {
                    "phase": "approaching",
                    "iterations": 1,
                    "distance_cm": distance_cm,
                }
                return SkillResult(
                    status="success",
                    message="Approche de la cible en boucle fermée (itération).",

                    break
                iterations += 1

                    self._stop_robot(context)
                    context.state["motion"] = "stop"
                    context.state["follow_target_state"] = {
                        "phase": "blocked",
                        "iterations": iterations,
                        "reason": "emergency_stop",
                    }
                    return SkillResult(
                        status="success",
                        message="Arrêt d'urgence demandé pendant le suivi.",
                        data={"emergency_stop": True, "iterations": iterations},
                        confidence=0.99,
                        next_action="stop_motion",
                    )

                if self._check_obstacle_fail_safe(context):
                    self._stop_robot(context)
                    context.state["motion"] = "stop"
                    context.state["follow_target_state"] = {
                        "phase": "blocked",
                        "iterations": iterations,
                        "reason": "obstacle_detected",
                    }
                    return SkillResult(
                        status="success",
                        message="Arrêt de sécurité : obstacle détecté pendant le suivi.",
                        data={"target_position": context.state.get("last_object_position"), "obstacle_detected": True},
                        confidence=0.99,
                        next_action="stop_motion",
                    )

                target_position, distance_cm = self._detect_target(context)
                current = now()

                if target_position is not None:
                    if phase == self.PHASE_SEARCH:
                        search_spent_s += current - phase_since
                        phase = self.PHASE_TRACK
                    last_seen_at = current
                    if target_position in {"left", "right"}:
                        sweep_direction = target_position

                    if distance_cm is not None and distance_cm <= reached_threshold_cm:

                            "iterations": iterations,

                                "distance_cm": distance_cm,
                                "close_enough": True,
                            },

                            next_action="stop_motion",

                        context.state["motion"] = f"align_{target_position}"
                        context.state["follow_target_state"] = {
                            "phase": "aligning",

                            "iterations": iterations,

                        self._pulse_motion(context, "forward", step_s)
                        context.state["motion"] = "approach_target"

                            "iterations": iterations,
                            "distance_cm": distance_cm,

                        status="failed",

                data={"error": str(exc), "skill": self.name},

from __future__ import annotations

import logging
import time
from collections.abc import Callable

from irobot.config.settings import RuntimeSettings, load_runtime_settings
from irobot.domain.skills.base_skill import Skill, SkillContext, SkillResult
from irobot.domain.skills.behaviors.target_tracking import TargetTrackingMixin


class FollowObjectSkill(TargetTrackingMixin, Skill):
    """Suit une cible jusqu'à un arrêt, ou pendant une durée explicitement demandée."""

    name = "follow_object"
    description = "Suit une cible visuelle à distance constante jusqu'à un arrêt"

    CONTROL_LOOP_PERIOD_S = 0.02
    DEFAULT_TARGET = "yellow"
    DEFAULT_STAND_OFF_CM = 100.0
    DEFAULT_TOLERANCE_CM = 15.0
    DEFAULT_DURATION_S = 60.0
    DEFAULT_MAX_TARGET_LOSS_S = 3.0
    ALIGN_STEP_S = 0.08
    FORWARD_MIN_S = 0.05
    FORWARD_MAX_S = 0.25
    BACKWARD_MIN_S = 0.05
    BACKWARD_MAX_S = 0.15

    def __init__(
        self,
        settings: RuntimeSettings | None = None,
        time_fn: Callable[[], float] = time.monotonic,
        sleep_fn: Callable[[float], None] = time.sleep,
    ) -> None:
        self._settings = settings
        self.time_fn = time_fn
        self.sleep_fn = sleep_fn

    @property
    def settings(self) -> RuntimeSettings:
        if self._settings is None:
            self._settings = load_runtime_settings()
        return self._settings

    def validate(self, context: SkillContext) -> bool:
        return context.robot is not None

    def _step(self, gap_cm: float, low_s: float, high_s: float, *, backward: bool = False) -> float:
        # Durée proportionnelle à l'écart, à la vitesse linéaire calibrée.
        calibrated_speed = self.settings.cm_per_second
        if backward and self.settings.backward_cm_per_second is not None:
            calibrated_speed = self.settings.backward_cm_per_second
        speed = max(1.0, float(calibrated_speed))
        return min(high_s, max(low_s, gap_cm / speed))

    def _result(
        self,
        context: SkillContext,
        status: str,
        message: str,
        data: dict,
        confidence: float,
    ) -> SkillResult:
        return SkillResult(
            status=status,
            message=message,
            data=data,
            confidence=confidence,
            next_action="stop_motion" if status == "success" else None,
        )

    def execute(self, context: SkillContext) -> SkillResult:
        try:
            return self._run(context)
        except Exception as exc:
            return SkillResult(
                status="failed",
                message=f"Erreur lors du suivi d'objet : {exc}",
                data={"error": str(exc), "skill": self.name},
                confidence=0.0,
                next_action=None,
            )
        finally:
            try:
                self._stop_robot(context)
            except Exception:
                logging.getLogger(__name__).exception("Impossible d'arreter le robot apres le suivi")

    def _run(self, context: SkillContext) -> SkillResult:
        params = dict(context.state.get("active_intent_parameters") or {})
        settings = self.settings
        if not callable(getattr(context.perception, "detect", None)):
            raise RuntimeError("Le suivi nécessite une perception visuelle active")
        if not params.get("target_object"):
            params["target_object"] = self.DEFAULT_TARGET
            context.state["active_intent_parameters"] = params

        approach_follow = context.state.get("active_intent_name") in {"follow_person", "follow_red_ball"}
        default_stand_off_cm = 20.0 if approach_follow else self.DEFAULT_STAND_OFF_CM
        default_tolerance_cm = 10.0 if approach_follow else self.DEFAULT_TOLERANCE_CM
        stand_off_cm = default_stand_off_cm if params.get("stand_off_cm") is None else min(
            max(self._to_float(params.get("stand_off_cm"), default_stand_off_cm), float(settings.min_stand_off_cm)),
            float(settings.max_stand_off_cm),
        )
        tolerance_cm = max(0.0, self._to_float(params.get("stand_off_tolerance_cm"), default_tolerance_cm))
        continuous = params.get("duration_s") is None
        duration_s = None if continuous else min(
            max(0.0, self._to_float(params.get("duration_s"), self.DEFAULT_DURATION_S)),
            float(settings.max_follow_duration_sec),
        )
        max_loss_s = max(0.0, self._to_float(params.get("max_target_loss_s"), self.DEFAULT_MAX_TARGET_LOSS_S))
        search_timeout_s = max(
            0.1, self._to_float(params.get("search_timeout_s"), float(settings.search_timeout_seconds))
        )

        now = self.time_fn
        started_at = now()
        last_seen_at = started_at
        search_started_at: float | None = None
        sweep_direction = "right"
        iterations = 0
        reacquisitions = 0
        final_distance_cm: float | None = None

        def set_state(phase: str, **extra) -> None:
            context.state["follow_target_state"] = {"phase": phase, "iterations": iterations, **extra}

        while duration_s is None or now() - started_at < duration_s:
            iterations += 1

            emergency = self._is_emergency_stop_requested(context)
            if emergency or self._check_obstacle_fail_safe(context):
                self._stop_robot(context)
                context.state["motion"] = "stop"
                set_state("blocked", reason="emergency_stop" if emergency else "obstacle_detected")
                data = {"obstacle_detected": True, "iterations": iterations, "final_distance_cm": final_distance_cm}
                if emergency:
                    data["emergency_stop"] = True
                return self._result(context, "success", "Arrêt de sécurité pendant le suivi.", data, 0.99)

            position, distance_cm = self._detect_target(context)
            current = now()

            if position is None:
                self._stop_robot(context)
                if current - last_seen_at < max_loss_s:
                    context.state["motion"] = "search_target"
                    set_state("searching", lost_for_s=round(current - last_seen_at, 3))
                    self.sleep_fn(self.CONTROL_LOOP_PERIOD_S)
                    continue
                if search_started_at is None:
                    search_started_at = current
                if not continuous and current - search_started_at >= search_timeout_s:
                    set_state("lost_target", searched_for_s=round(current - search_started_at, 3))
                    return self._result(
                        context,
                        "failed",
                        "Cible introuvable : recherche épuisée, arrêt.",
                        {
                            "target_lost": True,
                            "iterations": iterations,
                            "reacquisitions": reacquisitions,
                            "final_distance_cm": final_distance_cm,
                        },
                        0.1,
                    )
                self._sweep_step(context, sweep_direction, self.ALIGN_STEP_S, iterations)
                continue

            if search_started_at is not None:
                reacquisitions += 1
                search_started_at = None
            last_seen_at = current
            if distance_cm is not None:
                final_distance_cm = distance_cm

            if position in {"left", "right"}:
                sweep_direction = position
                self._pulse_motion(context, position, self.ALIGN_STEP_S)
                context.state["motion"] = f"align_{position}"
                set_state("aligning", direction=position)
            elif distance_cm is not None and distance_cm > stand_off_cm + tolerance_cm:
                self._pulse_motion(
                    context, "forward", self._step(distance_cm - stand_off_cm, self.FORWARD_MIN_S, self.FORWARD_MAX_S)
                )
                context.state["motion"] = "follow_object"
                set_state("approaching", distance_cm=distance_cm)
            elif not approach_follow and distance_cm is not None and distance_cm < stand_off_cm - tolerance_cm:
                self._pulse_motion(
                    context, "backward",
                    self._step(stand_off_cm - distance_cm, self.BACKWARD_MIN_S, self.BACKWARD_MAX_S, backward=True),
                )
                context.state["motion"] = "follow_object"
                set_state("backing", distance_cm=distance_cm)
            else:
                self._stop_robot(context)
                context.state["motion"] = "stop"
                set_state("holding", distance_cm=distance_cm)
            self.sleep_fn(self.CONTROL_LOOP_PERIOD_S)

        self._stop_robot(context)
        context.state["motion"] = "stop"
        assert duration_s is not None
        return self._result(
            context,
            "success",
            f"Suivi terminé après {duration_s:.0f} s.",
            {
                "iterations": iterations,
                "duration_s": duration_s,
                "final_distance_cm": final_distance_cm,
                "reacquisitions": reacquisitions,
            },
            0.9,
        )

from __future__ import annotations

from collections.abc import Callable
from typing import Any

from irobot.domain.skills.base_skill import SkillContext


FOLLOW_INTENTS = frozenset({"follow_person", "follow_red_ball", "follow_object"})


class TargetTrackingMixin:
    """Briques communes aux behaviors de suivi visuel (détection, sécurité, balayage)."""

    SEARCH_SETTLE_S = 0.15

    sleep_fn: Callable[[float], None]

    @staticmethod
    def _to_float(value: Any, default: float) -> float:
        try:
            return float(value)
        except (TypeError, ValueError):
            return default

    @staticmethod
    def _normalize_position(position: Any) -> str | None:
        if position is None:
            return None
        normalized = str(position).strip().lower()
        if normalized in {"center", "centre", "middle"}:
            return "center"
        if normalized in {"left", "right"}:
            return normalized
        return None

    def _check_obstacle_fail_safe(self, context: SkillContext) -> bool:
        if bool(context.state.get("obstacle_detected", False)):
            return True

        checker = getattr(context.robot, "check_obstacle", None)
        if checker is None:
            return False

        try:
            reading = checker()
        except Exception:
            return True

        if reading is None:
            return True
        return bool(reading)

    def _is_emergency_stop_requested(self, context: SkillContext) -> bool:
        robot = context.robot
        detector = getattr(robot, "_is_emergency_stop_requested", None) or getattr(
            robot, "is_emergency_stop_requested", None
        )
        if not callable(detector):
            return False
        try:
            return bool(detector())
        except Exception:
            return True

    def _stop_robot(self, context: SkillContext) -> None:
        stop = getattr(context.robot, "stop", None)
        if callable(stop):
            stop()

    def _pulse_motion(self, context: SkillContext, direction: str, duration_s: float) -> None:
        duration = max(0.0, float(duration_s))
        directional_for = getattr(context.robot, f"{direction}_for", None)
        if callable(directional_for):
            directional_for(duration)
            return

        directional = getattr(context.robot, direction, None)
        if callable(directional):
            directional()
            if duration > 0.0:
                self.sleep_fn(duration)

    def _sweep_step(
        self,
        context: SkillContext,
        direction: str,
        step_s: float,
        iterations: int,
    ) -> None:
        """Une impulsion de balayage de recherche suivie d'un temps de stabilisation."""
        self._pulse_motion(context, direction, step_s)
        context.state["motion"] = "search_target"
        context.state["follow_target_state"] = {
            "phase": "searching",
            "direction": direction,
            "iterations": iterations,
        }
        self.sleep_fn(self.SEARCH_SETTLE_S)

    def _detect_target(self, context: SkillContext) -> tuple[str | None, float | None]:
        detector = getattr(context.perception, "detect", None)

        if callable(detector) and context.perception_target is not None:
            detection = detector(target=context.perception_target)
            if detection is None:
                context.state["last_object_position"] = None
                context.state["last_object_distance_cm"] = None
                context.state["last_ball_position"] = None
            else:
                context.state["target_object"] = getattr(detection, "label", None)
                context.state["last_object_position"] = getattr(detection, "position", None)
                context.state["last_object_distance_cm"] = getattr(detection, "distance_cm", None)

        raw_position = context.state.get("last_object_position") or context.state.get("last_ball_position")
        position = self._normalize_position(raw_position)

        distance_raw = context.state.get("last_object_distance_cm")
        try:
            distance_cm = float(distance_raw) if distance_raw is not None else None
        except (TypeError, ValueError):
            distance_cm = None
        return position, distance_cm

from __future__ import annotations

from collections.abc import Iterator, Mapping, MutableMapping
from dataclasses import dataclass, field
from typing import Any, ClassVar


@dataclass
class BlackboardState(MutableMapping[str, Any]):
    """État partagé typé entre perception, planification et action."""

    active_intent: Any | None = None
    active_intent_name: str | None = None
    active_intent_parameters: dict[str, Any] = field(default_factory=dict)
    active_skill: str | None = None

    camera_ready: bool = False
    target_object: str | None = None
    last_camera_position: Any | None = None
    last_object_position: str | None = None
    last_ball_position: str | None = None
    last_object_distance_cm: float | None = None
    obstacle_detected: bool = False
    close_enough: bool = False

    motion: str | None = None
    follow_target_state: dict[str, Any] | None = None
    last_skill_result: Any | None = None

    extra: dict[str, Any] = field(default_factory=dict)

    KNOWN_FIELDS: ClassVar[set[str]] = {
        "active_intent",
        "active_intent_name",
        "active_intent_parameters",
        "active_skill",
        "camera_ready",
        "target_object",
        "last_camera_position",
        "last_object_position",
        "last_ball_position",
        "last_object_distance_cm",
        "obstacle_detected",
        "close_enough",
        "motion",
        "follow_target_state",
        "last_skill_result",
    }

    @classmethod
    def from_mapping(cls, data: Mapping[str, Any] | None) -> "BlackboardState":
        state = cls()
        if not data:
            return state
        for key, value in data.items():
            state[key] = value
        return state

    def __getitem__(self, key: str) -> Any:
        if key in self.KNOWN_FIELDS:
            return getattr(self, key)
        return self.extra[key]

    def __setitem__(self, key: str, value: Any) -> None:
        if key in self.KNOWN_FIELDS:
            setattr(self, key, value)
            return
        self.extra[key] = value

    def __delitem__(self, key: str) -> None:
        if key in self.KNOWN_FIELDS:
            setattr(self, key, None)
            return
        del self.extra[key]

    def __iter__(self) -> Iterator[str]:
        yield from self.KNOWN_FIELDS
        yield from self.extra

    def __len__(self) -> int:
        return len(self.KNOWN_FIELDS) + len(self.extra)

    def as_dict(self) -> dict[str, Any]:
        payload = {key: getattr(self, key) for key in self.KNOWN_FIELDS}
        payload.update(self.extra)
        return payload

from irobot.domain.skills.base_skill import Skill, SkillContext, SkillResult


class DetectObjectSkill(Skill):
    """Skill de perception : détecte un objet cible dans le champ visuel."""

    name = "detect_object"
    description = "Retourne la position d'un objet cible si elle est visible"

    def validate(self, context: SkillContext) -> bool:

            position = context.state.get("last_object_position")

            distance_cm = context.state.get("last_object_distance_cm")

                message=f"Erreur lors de la détection : {exc}",
                data={"error": str(exc), "skill": self.name},

from __future__ import annotations

from typing import Any

from irobot.config.settings import load_runtime_settings
from irobot.domain.skills.base_skill import Skill, SkillContext, SkillResult

SETTINGS = load_runtime_settings()

_METER_UNITS = {"m", "meter", "meters", "metre", "metres", "mètre", "mètres"}


def normalize_distance_unit(unit: Any) -> str:
    """Ramène une unité libre à "m" ou "cm" (inconnue -> "cm")."""
    text = str(unit if unit is not None else "cm").strip().lower()
    if text in _METER_UNITS:
        return "m"
    return "cm"


class BaseMotionPrimitiveSkill(Skill):
    """Classe de base factorisée pour les primitives de mouvement."""

    motion: str = "motion"
    command_method: str = ""
    command_for_method: str = ""
    parameter_mode: str = "distance"  # distance|angle
    success_message: str = "Mouvement exécuté."
    failure_message: str = "Le robot ne fournit pas la méthode requise"
    error_prefix: str = "Erreur lors du mouvement"

    def validate(self, context: SkillContext) -> bool:
        return context.robot is not None

    @staticmethod
    def _to_float(value: Any) -> float | None:
        try:
            return float(value)
        except (TypeError, ValueError):
            return None

    def _resolve_motion_parameters(self, context: SkillContext) -> dict[str, Any]:
        params = dict(context.state.get("active_intent_parameters") or {})

        if self.parameter_mode == "angle":
            angle = params.get("angle")
            if angle is None and params.get("full_rotation"):
                angle = 360
            angle_value = self._to_float(angle)
            duration_value = self._to_float(params.get("duration"))
            if duration_value is None and angle_value is not None:
                duration_value = max(0.1, angle_value / SETTINGS.deg_per_second)
            return {
                "angle": angle_value,
                "duration": duration_value,
            }

        distance = params.get("distance", params.get("side"))
        unit = normalize_distance_unit(params.get("unit")) if distance is not None else None
        distance_value = self._to_float(distance)
        duration_value = self._to_float(params.get("duration"))
        if duration_value is None and distance_value is not None:
            distance_cm = distance_value * 100.0 if unit == "m" else distance_value
            speed = SETTINGS.cm_per_second
            if self.motion == "backward" and SETTINGS.backward_cm_per_second is not None:
                speed = SETTINGS.backward_cm_per_second
            duration_value = max(0.1, distance_cm / speed)
        return {
            "distance": distance_value,
            "unit": unit,
            "duration": duration_value,
        }

    def _execute_motion(self, context: SkillContext, duration_value: float | None) -> tuple[bool, str | None]:
        command = getattr(context.robot, self.command_method, None)
        if not callable(command):
            return False, None

        command_for = getattr(context.robot, self.command_for_method, None)
        previous_speed = getattr(context.robot, "speed", None)
        set_speed = getattr(context.robot, "set_speed", None)
        use_turn_speed = self.parameter_mode == "angle" and callable(set_speed)
        if use_turn_speed:
            set_speed(SETTINGS.turn_speed)
        stop_reason = None
        try:
            if duration_value is not None and callable(command_for):
                outcome = command_for(duration_value)
                if isinstance(outcome, str) and outcome in {"obstacle", "emergency_stop"}:
                    stop_reason = outcome
            else:
                command()
        finally:
            if use_turn_speed and previous_speed is not None and duration_value is not None:
                set_speed(previous_speed)
        return True, stop_reason

    def execute(self, context: SkillContext) -> SkillResult:
        try:
            payload = self._resolve_motion_parameters(context)
            duration_value = self._to_float(payload.get("duration"))

            executed, stop_reason = self._execute_motion(context, duration_value)
            if not executed:
                return SkillResult(
                    status="failed",
                    message=self.failure_message,
                    data={"skill": self.name},
                    confidence=0.0,
                    next_action=None,
                )

            if stop_reason is not None:
                context.state["motion"] = "stop"
                return SkillResult(
                    status="success",
                    message="Mouvement interrompu par un arrêt de sécurité.",
                    data={
                        **payload,
                        "motion": "stop",
                        "obstacle_detected": stop_reason == "obstacle",
                        "emergency_stop": stop_reason == "emergency_stop",
                    },
                    confidence=1.0,
                    next_action="stop_motion",
                )

            context.state["motion"] = self.motion
            return SkillResult(
                status="success",
                message=self.success_message,
                data={"motion": self.motion, **payload},
                confidence=1.0,
                next_action="track_target",
            )
        except Exception as exc:  # pragma: no cover - logique métier
            return SkillResult(
                status="failed",
                message=f"{self.error_prefix} : {exc}",
                data={"error": str(exc), "skill": self.name},
                confidence=0.0,
                next_action=None,
            )

from __future__ import annotations

import re
import subprocess

CARD_NAME = "wm8960"


def _run_command(args: list[str]) -> str | None:
    try:
        proc = subprocess.run(args, capture_output=True, text=True, timeout=10)
    except (FileNotFoundError, subprocess.TimeoutExpired):
        return None
    if proc.returncode != 0:
        return None
    return proc.stdout


def find_wm8960_card() -> int | None:
    out = _run_command(["arecord", "-l"])
    if out is None:
        return None
    for line in out.splitlines():
        match = re.search(r"card (\d+):\s*(\S+)", line)
        if match and CARD_NAME in match.group(2).lower():
            return int(match.group(1))
    return None


def detect_bluetooth_input_device() -> str | None:
    for cmd in (["pactl", "list", "short", "sources"], ["pacmd", "list-sources"], ["arecord", "-L"]):
        out = _run_command(cmd)
        if out is None:
            continue
        for line in out.splitlines():
            lowered = line.lower()
            if any(token in lowered for token in ("bluez", "bluetooth", "headset", "hfp")):
                return "default"
    return None


def resolve_input_device(device_name: str | None) -> str:
    if device_name and device_name != "auto":
        if device_name == "wm8960":
            card = find_wm8960_card()
            if card is not None:
                return f"plughw:{card},0"
            raise RuntimeError("Aucune carte WM8960 détectée.")
        if device_name == "bluetooth":
            bt = detect_bluetooth_input_device()
            if bt is not None:
                return bt
            raise RuntimeError("Aucun casque Bluetooth détecté comme source audio.")
        return device_name

    bt = detect_bluetooth_input_device()
    if bt is not None:
        return bt

    card = find_wm8960_card()
    if card is not None:
        return f"plughw:{card},0"

    return "default"

"""ALSA capture and lazy Vosk transcription; no planning or motor logic."""

from contextlib import contextmanager
import json
import os
import subprocess

RATE = 16000
MODEL_PATH = os.path.expanduser("~/models/vosk-model-small-fr-0.22")


@contextmanager


def _arecord_stream(device: str):
    """Ouvre un flux arecord et garantit son arrêt même en cas d'exception."""
    args = [
        "arecord",
        "-D",
        device,
        "-f",
        "S16_LE",
        "-r",
        str(RATE),
        "-c",
        "1",
        "-t",
        "raw",
    ]
    proc = subprocess.Popen(args, stdout=subprocess.PIPE)
    try:
        if proc.stdout is None:
            raise RuntimeError("Impossible d'ouvrir le flux audio arecord.")
        yield proc.stdout
    finally:
        if proc.poll() is None:
            proc.terminate()
            try:
                proc.wait(timeout=1.0)
            except subprocess.TimeoutExpired:
                proc.kill()
                proc.wait(timeout=1.0)
        if proc.stdout is not None:
            proc.stdout.close()


def transcribe_loop(device: str):
    """Lit le flux brut du microphone et le transmet à Vosk."""
    try:
        import vosk
    except ModuleNotFoundError as exc:
        raise RuntimeError("La dépendance 'vosk' est absente. Installez les paquets Raspberry Pi requis.") from exc

    model = vosk.Model(MODEL_PATH)
    recognizer = vosk.KaldiRecognizer(model, RATE)
    print(f"(parle dans le micro — device: {device} — Ctrl+C pour quitter)\n", flush=True)

    bytes_per_chunk = 8000
    with _arecord_stream(device) as audio_stream:
        while True:
            raw = audio_stream.read(bytes_per_chunk)
            if not raw:
                break
            if recognizer.AcceptWaveform(raw):
                result = json.loads(recognizer.Result())
                text = result.get("text", "").strip()
                if text:
                    yield text
                    recognizer.Reset()

from __future__ import annotations

import dataclasses
import logging
import threading

import cv2
import numpy as np
from numpy.typing import NDArray

try:
    from picamera2 import Picamera2
except ModuleNotFoundError:  # pragma: no cover - non-Raspberry environments
    Picamera2 = None
from irobot.config.settings import load_runtime_settings
from irobot.domain.perception import PerceptionTarget
from irobot.hardware.camera.detector.target_detector import CameraTargetDetector
from irobot.hardware.camera.frame_enhancer import FrameEnhancer, configure_exposure
from irobot.interfaces.perception_contracts import (
    ObjectDetection,
    ObjectPerceptionPipeline,
    TargetFrameDetector,
)

logger = logging.getLogger(__name__)


class ExponentialSmoother:
    """Filtre exponentiel : s = alpha * x + (1 - alpha) * s."""

    def __init__(self, alpha: float = 0.4) -> None:
        if not 0.0 < alpha <= 1.0:
            raise ValueError("alpha doit être dans ]0, 1]")
        self.alpha = float(alpha)
        self._value: float | None = None

    def update(self, measure: float | None) -> float | None:
        """Ajoute une mesure ; une mesure None réinitialise le filtre (cible perdue)."""
        if measure is None:
            self._value = None
            return None
        if self._value is None:
            self._value = float(measure)
        else:
            self._value = self.alpha * float(measure) + (1.0 - self.alpha) * self._value
        return self._value

    def reset(self) -> None:
        self._value = None


class CameraLivePerceptionPipeline(ObjectPerceptionPipeline):
    """Perception Picamera2 : balles, ArUco (marker/me) et YuNet (face/human).

    L'identité du marqueur configurée est utilisée, sinon aruco/DICT_4X4_50/1.
    La distance est estimée par pose pour un carré de 8 cm, avec la focale
    configurée et sans correction de distorsion.
    Les seuils horizontaux sont rapportés à une largeur de 480 px.
    """

    TARGET_FRAME_TIMEOUT_S = 1.0

    def __init__(
        self,
        *,
        camera_index: int = 0,
        poll_interval_s: float = 0.05,
        ball_diameter_cm: float | None = None,
        focal_px: float | None = None,
        smoothing_alpha: float = 0.4,
        detector: TargetFrameDetector | None = None,
        sensor: object | None = None,
        autostart: bool = True,
    ) -> None:
        self.camera_index = int(camera_index)
        self.sensor = sensor
        self.poll_interval_s = max(0.0, float(poll_interval_s))

        settings = load_runtime_settings()
        self._settings = settings
        self.ball_diameter_cm = float(settings.camera_ball_diameter_cm if ball_diameter_cm is None else ball_diameter_cm)
        self.focal_px = float(settings.camera_focal_px if focal_px is None else focal_px)
        self._detector = detector or CameraTargetDetector(
            settings,
            ball_diameter_cm=self.ball_diameter_cm,
            focal_px=self.focal_px,
        )
        self._frame_enhancer = FrameEnhancer(settings)
        self._distance_smoother = ExponentialSmoother(smoothing_alpha)
        self._active_target: PerceptionTarget | None = None
        self._last_detection: ObjectDetection | None = None
        self._requested_target: PerceptionTarget | None = None
        self._lock = threading.Lock()
        self._frame_condition = threading.Condition(self._lock)
        self._target_revision = 0
        self._frame_revision = -1
        self._capture_finished = False
        self._stop_event = threading.Event()
        self._thread = threading.Thread(target=self._capture_loop, name="camera-live-perception", daemon=False)
        if autostart:
            self._thread.start()

    def _process_frame(
        self,
        frame_rgb: NDArray[np.uint8],
        *,
        target: PerceptionTarget,
    ) -> ObjectDetection | None:
        if target is not self._active_target:
            if self._active_target is not None:
                self._detector.reset(self._active_target)
            self._detector.reset(target)
            self._active_target = target

        enhanced = self._frame_enhancer.apply(frame_rgb)
        detection = self._detector.detect(enhanced, target)
        if detection is None:
            self._distance_smoother.reset()
            return None

        distance = self._distance_smoother.update(
            detection.distance_visual_cm
            if detection.distance_visual_cm is not None
            else detection.distance_cm
        )
        return dataclasses.replace(
            detection,
            distance_visual_cm=distance,
            distance_cm=distance,
        )

    def _capture_loop(self) -> None:
        if Picamera2 is None:
            logger.error("Perception caméra indisponible : Picamera2 absent.")
            with self._frame_condition:
                self._capture_finished = True
                self._frame_condition.notify_all()
            return

        camera = None
        try:
            camera = Picamera2(self.camera_index)
            # Picamera2 BGR888 fournit des octets R, G, B à _process_frame.
            config = camera.create_preview_configuration(
                main={"size": self._settings.camera_capture_size_px, "format": "BGR888"}
            )
            camera.configure(config)
            configure_exposure(camera, self._settings)
            camera.start()
            processed_revision = -1

            while not self._stop_event.is_set():
                with self._lock:
                    target = self._requested_target
                    revision = self._target_revision
                frame_rgb = camera.capture_array()
                if revision != processed_revision:
                    if self._active_target is not None:
                        self._detector.reset(self._active_target)
                    self._active_target = None
                    self._distance_smoother.reset()
                    processed_revision = revision
                rotated_frame_rgb = cv2.rotate(frame_rgb, cv2.ROTATE_90_CLOCKWISE)
                detection = (
                    self._process_frame(rotated_frame_rgb, target=target)
                    if target is not None
                    else None
                )
                with self._frame_condition:
                    # Une image en vol ne doit pas satisfaire une nouvelle demande de cible.
                    if revision == self._target_revision:
                        self._last_detection = detection
                        self._frame_revision = revision
                    self._frame_condition.notify_all()

                if self.poll_interval_s > 0.0:
                    self._stop_event.wait(self.poll_interval_s)
        except Exception:
            logger.exception("Échec de la capture caméra.")
            with self._frame_condition:
                self._last_detection = None
                self._frame_revision = -1
        finally:
            with self._frame_condition:
                self._capture_finished = True
                self._frame_condition.notify_all()
            if camera is not None:
                try:
                    camera.stop()
                except Exception:
                    logger.exception("Échec de l'arrêt de la caméra.")
                try:
                    camera.close()
                except Exception:
                    logger.exception("Échec de la fermeture de la caméra.")

    def detect(self, *, target: PerceptionTarget) -> ObjectDetection | None:
        with self._frame_condition:
            if target is not self._requested_target:
                self._target_revision += 1
                self._last_detection = None
            self._requested_target = target
            revision = self._target_revision
            if self._thread.is_alive():
                ready = self._frame_condition.wait_for(
                    lambda: self._frame_revision == revision
                    or self._stop_event.is_set()
                    or self._capture_finished,
                    timeout=self.TARGET_FRAME_TIMEOUT_S,
                )
                if not ready:
                    logger.error(
                        "Délai dépassé en attendant une image caméra pour la cible %s.",
                        target.value,
                    )
                    return None
                if self._frame_revision != revision:
                    return None
            detection = self._last_detection

        if detection is None:
            return None
        return dataclasses.replace(detection)

    def close(self, timeout_s: float | None = None) -> None:
        with self._frame_condition:
            self._stop_event.set()
            self._frame_condition.notify_all()
        if self._thread.is_alive():
            self._thread.join(timeout=None if timeout_s is None else max(0.0, float(timeout_s)))
            if self._thread.is_alive():
                raise TimeoutError("Le thread caméra n'est pas arrêté.")

logger = logging.getLogger(__name__)

MARKER_SIDE_CM = 8.0


def estimate_marker_distance_cm(
    corners_px: tuple[tuple[float, float], ...],
    focal_px: float,
    frame_size_px: tuple[int, int] | None = None,
    oriented_frame_size_px: tuple[int, int] | None = None,
) -> float | None:
    """Estimate direct range from pose, assuming centred optics without distortion."""
    if frame_size_px is None or oriented_frame_size_px is None:
        settings = load_runtime_settings()
        if oriented_frame_size_px is None:
            oriented_frame_size_px = settings.camera_oriented_frame_size_px
        if frame_size_px is None:
            frame_size_px = oriented_frame_size_px
    assert frame_size_px is not None
    assert oriented_frame_size_px is not None
    width, height = frame_size_px
    reference_width, reference_height = oriented_frame_size_px
    if width <= 0 or height <= 0 or reference_width <= 0 or reference_height <= 0:
        raise ValueError("Frame dimensions must be positive")
    corners = np.asarray(corners_px, dtype=np.float64)
    if corners.shape != (4, 2) or not np.isfinite(corners).all():
        return None
    contour = corners.astype(np.float32)
    if not cv2.isContourConvex(contour) or cv2.contourArea(contour) <= 0:
        return None
    half_side = MARKER_SIDE_CM / 2
    object_points = np.array(
        [
            (-half_side, -half_side, 0),
            (half_side, -half_side, 0),
            (half_side, half_side, 0),
            (-half_side, half_side, 0),
        ],
        dtype=np.float64,
    )
    camera_matrix = np.array(
        [
            (focal_px * width / reference_width, 0, width / 2),
            (0, focal_px * height / reference_height, height / 2),
            (0, 0, 1),
        ],
        dtype=np.float64,
    )
    try:
        success, _, translation = cv2.solvePnP(
            object_points, corners, camera_matrix, np.zeros(5),
            flags=cv2.SOLVEPNP_ITERATIVE,
        )
    except cv2.error:
        logger.exception("Échec de l'estimation de pose ArUco.")
        return None
    if not success or not np.isfinite(translation).all() or translation[2, 0] <= 0:
        return None
    return float(np.linalg.norm(translation))


class ObservationDisposition(str, Enum):
    VALID = "valid"
    ABSENT = "absent"
    AMBIGUOUS = "ambiguous"
    REJECTED = "rejected"
    FAULT = "fault"


@dataclass(frozen=True, slots=True)
class ArucoFrameDetection:
    disposition: ObservationDisposition
    identity: MarkerIdentity | None = None
    corners_px: tuple[tuple[float, float], ...] | None = None
    reason: str | None = None
    distance_cm: float | None = None
    position: str = "UNKNOWN"
    bbox_px: tuple[int, int, int, int] | None = None
    confidence: float = 0.0

    def as_object_detection(self) -> ObjectDetection | None:
        if self.disposition is not ObservationDisposition.VALID or self.distance_cm is None:
            return None
        return ObjectDetection(
            label="marker_1",
            position=self.position,
            distance_cm=self.distance_cm,
            distance_visual_cm=self.distance_cm,
            confidence=self.confidence,
            bbox_px=self.bbox_px,
        )


class MarkerDetectorBackend(Protocol):
    def detectMarkers(self, image: object) -> tuple[object, object, object]: ...


class OpenCVMarkerDetector(Detector[ArucoFrameDetection]):
    """Detect one configured ArUco ID from RGB frames."""

    def __init__(
        self,
        identity: MarkerIdentity,
        *,
        focal_px: float,
        aruco_api: object | None = None,
        oriented_frame_size_px: tuple[int, int] | None = None,
    ) -> None:
        if identity.family != "aruco":
            raise ValueError("The configured marker family is not ArUco")
        if not isfinite(focal_px) or focal_px <= 0:
            raise ValueError("Camera focal length must be finite and positive")
        if (not identity.identifier.isascii() or not identity.identifier.isdigit()
                or str(int(identity.identifier)) != identity.identifier):
            raise ValueError("ArUco identifier must be a canonical non-negative integer")
        if aruco_api is None:
            aruco_api = self._load_aruco_api()
        if aruco_api is None:
            raise RuntimeError("OpenCV ArUco API is unavailable; no fallback detector is selected")
        get_dictionary = getattr(aruco_api, "getPredefinedDictionary", None)
        parameters_factory = getattr(aruco_api, "DetectorParameters", None)
        detector_factory = getattr(aruco_api, "ArucoDetector", None)
        dictionary_code = getattr(aruco_api, identity.dictionary, None)
        if (not callable(get_dictionary) or not callable(parameters_factory)
                or not callable(detector_factory) or isinstance(dictionary_code, bool)
                or not isinstance(dictionary_code, int)):
            raise ValueError(f"Unsupported OpenCV ArUco dictionary/API: {identity.dictionary}")
        dictionary = get_dictionary(dictionary_code)
        detector = detector_factory(dictionary, parameters_factory())
        if not callable(getattr(detector, "detectMarkers", None)):
            raise RuntimeError("OpenCV ArUco detector lacks detectMarkers")
        self.identity = identity
        self.expected_id = int(identity.identifier)
        self.focal_px = float(focal_px)
        self.oriented_frame_size_px = (
            oriented_frame_size_px or load_runtime_settings().camera_oriented_frame_size_px
        )
        self._detector: MarkerDetectorBackend = detector

    @staticmethod
    def _load_aruco_api() -> object | None:
        import cv2

        return getattr(cv2, "aruco", None)

    def detect(self, frame_rgb: NDArray[np.uint8]) -> ArucoFrameDetection:
        grayscale = cv2.cvtColor(frame_rgb, cv2.COLOR_RGB2GRAY)
        corners_raw, ids_raw, _ = self._detector.detectMarkers(grayscale)
        corners = self._as_python_values(corners_raw)
        marker_ids = self._flatten_ids(ids_raw)
        if marker_ids is None:
            logger.warning("Détection ArUco non valide : malformed_marker_ids")
            return ArucoFrameDetection(
                ObservationDisposition.FAULT, reason="malformed_marker_ids",
            )
        if not marker_ids:
            if corners not in (None, []):
                logger.warning("Détection ArUco non valide : corners_without_marker_ids")
                return ArucoFrameDetection(
                    ObservationDisposition.FAULT, reason="corners_without_marker_ids",
                )
            return ArucoFrameDetection(ObservationDisposition.ABSENT)
        if not isinstance(corners, list) or len(corners) != len(marker_ids):
            logger.warning("Détection ArUco non valide : marker_id_corner_count_mismatch")
            return ArucoFrameDetection(
                ObservationDisposition.FAULT, reason="marker_id_corner_count_mismatch",
            )
        matches = [index for index, marker_id in enumerate(marker_ids) if marker_id == self.expected_id]
        if not matches:
            return ArucoFrameDetection(ObservationDisposition.ABSENT)
        if len(matches) > 1:
            return ArucoFrameDetection(
                ObservationDisposition.AMBIGUOUS, identity=self.identity,
                reason="multiple_configured_markers",
            )
        parsed_corners = self._parse_corners(corners[matches[0]])
        if parsed_corners is None:
            logger.warning("Détection ArUco non valide : malformed_marker_corners")
            return ArucoFrameDetection(
                ObservationDisposition.REJECTED, identity=self.identity,
                reason="malformed_marker_corners",
            )
        distance = estimate_marker_distance_cm(
            parsed_corners,
            self.focal_px,
            (frame_rgb.shape[1], frame_rgb.shape[0]),
            self.oriented_frame_size_px,
        )
        center_x = sum(x for x, _ in parsed_corners) / 4
        frame_width = frame_rgb.shape[1]
        position = (
            "left" if center_x < frame_width * 3 / 8
            else "right" if center_x > frame_width * 5 / 8
            else "center"
        )
        left = round(min(x for x, _ in parsed_corners))
        top = round(min(y for _, y in parsed_corners))
        right = round(max(x for x, _ in parsed_corners))
        bottom = round(max(y for _, y in parsed_corners))
        if distance is None:
            logger.warning("Pose ArUco inexploitable ; distance indisponible.")
        return ArucoFrameDetection(
            ObservationDisposition.VALID,
            identity=self.identity,
            corners_px=parsed_corners,
            distance_cm=distance,
            position=position,
            bbox_px=(left, top, right - left, bottom - top),
            confidence=1.0,
        )

    @classmethod
    def _as_python_values(cls, value: object) -> object:
        tolist = getattr(value, "tolist", None)
        if callable(tolist):
            return cls._as_python_values(tolist())
        if isinstance(value, (tuple, list)):
            return [cls._as_python_values(item) for item in value]
        return value

    @classmethod
    def _flatten_ids(cls, values: object) -> list[int] | None:
        normalized = cls._as_python_values(values)
        if normalized is None:
            return []
        flattened: list[int] = []

        def append(value: object) -> bool:
            if isinstance(value, list):
                return all(append(item) for item in value)
            if isinstance(value, int) and not isinstance(value, bool):
                flattened.append(value)
                return True
            return False

        return flattened if append(normalized) else None

    @classmethod
    def _parse_corners(cls, value: object) -> tuple[tuple[float, float], ...] | None:
        points = cls._as_python_values(value)
        while isinstance(points, list) and len(points) == 1 and isinstance(points[0], list):
            points = points[0]
        if not isinstance(points, list) or len(points) != 4:
            return None
        parsed: list[tuple[float, float]] = []
        for point in points:
            if not isinstance(point, list) or len(point) != 2:
                return None
            x, y = point
            if (isinstance(x, bool) or not isinstance(x, (int, float))
                    or isinstance(y, bool) or not isinstance(y, (int, float))
                    or not isfinite(x) or not isfinite(y)):
                return None
            parsed.append((float(x), float(y)))
        return tuple(parsed)

from abc import ABC, abstractmethod

import numpy as np

class Detector(ABC, Generic[DetectionT]):

        """Return a detection result for the given RGB frame."""

"""Détection des balles rouges ou jaunes dans une image."""

import math
from typing import TypedDict

import cv2
import numpy as np
from numpy.typing import NDArray

from .base_detector import Detector
from .distance_estimator import estimate_distance_cm
from irobot.interfaces.perception_contracts import ObjectDetection


class BallCandidate(TypedDict):
    bbox: tuple[int, int, int, int]
    area: float
    diameter_px: float


MAX_AREA_RATIO = 0.25
BORDER_MARGIN = 10

COLOR_CONFIG = {
    "red": {
        "hsv_min": np.array((0, 180, 80)),
        "hsv_max": np.array((8, 255, 255)),
        "hsv_min_2": np.array((172, 180, 80)),
        "hsv_max_2": np.array((179, 255, 255)),
        "min_area": 700,
        "min_circularity": 0.65,
        "min_extent": 0.30,
        "min_solidity": 0.80,
        "aspect_min": 0.75,
        "aspect_max": 1.30,
    },
    "yellow": {
        "hsv_min": np.array((23, 150, 80)),
        "hsv_max": np.array((35, 255, 255)),
        "min_area": 700,
        "min_circularity": 0.65,
        "min_extent": 0.30,
        "min_solidity": 0.80,
        "aspect_min": 0.75,
        "aspect_max": 1.30,
    },
}


def build_mask_for_color(hsv: NDArray[np.uint8], color_name: str) -> NDArray[np.uint8]:
    """Construit le masque HSV de la couleur demandée."""
    cfg = COLOR_CONFIG[color_name]

    if color_name == "red":
        mask1 = cv2.inRange(hsv, cfg["hsv_min"], cfg["hsv_max"])
        mask2 = cv2.inRange(hsv, cfg["hsv_min_2"], cfg["hsv_max_2"])
        return cv2.bitwise_or(mask1, mask2)

    if color_name == "yellow":
        return cv2.inRange(hsv, cfg["hsv_min"], cfg["hsv_max"])

    raise ValueError(f"Couleur non supportée : {color_name}")


def classify_colored_contour(
    contour: NDArray[np.int32], frame: NDArray[np.uint8], color_name: str,
) -> bool:
    """Indique si un contour coloré correspond à une balle."""
    area = cv2.contourArea(contour)
    x, y, width, height = cv2.boundingRect(contour)
    if x <= BORDER_MARGIN or y <= BORDER_MARGIN:
        return False
    if x + width >= frame.shape[1] - BORDER_MARGIN or y + height >= frame.shape[0] - BORDER_MARGIN:
        return False
    if area > frame.shape[0] * frame.shape[1] * MAX_AREA_RATIO:
        return False

    perimeter = cv2.arcLength(contour, True)
    if perimeter == 0:
        return False

    circularity = (4 * np.pi * area) / (perimeter * perimeter)
    aspect = width / height if height else 0.0
    extent = area / (width * height) if width * height else 0.0
    hull_area = cv2.contourArea(cv2.convexHull(contour))
    solidity = area / hull_area if hull_area > 0 else 0.0

    cfg = COLOR_CONFIG[color_name]
    return (
        area >= cfg["min_area"]
        and circularity >= cfg["min_circularity"]
        and cfg["aspect_min"] <= aspect <= cfg["aspect_max"]
        and extent >= cfg["min_extent"]
        and solidity >= cfg["min_solidity"]
    )


def detect_colored_balls(
    frame: NDArray[np.uint8], color_name: str = "red", *, is_rgb: bool = False,
) -> list[BallCandidate]:
    """Détecte les balles et expose leur boîte, aire et diamètre équivalent."""
    hsv = cv2.cvtColor(frame, cv2.COLOR_RGB2HSV if is_rgb else cv2.COLOR_BGR2HSV)
    mask = build_mask_for_color(hsv, color_name)
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, np.ones((5, 5), np.uint8))
    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, np.ones((9, 9), np.uint8))
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    candidates: list[BallCandidate] = []
    for contour in contours:
        if not classify_colored_contour(contour, frame, color_name):
            continue

        x, y, width, height = cv2.boundingRect(contour)
        area = cv2.contourArea(contour)
        candidates.append({
            "bbox": (x, y, width, height),
            "area": area,
            "diameter_px": float(np.sqrt(4.0 * area / np.pi)),
        })
    return candidates


def select_largest_ball(candidates: list[BallCandidate]) -> BallCandidate | None:
    """Choisit la plus grande balle détectée."""
    return max(candidates, key=lambda candidate: candidate["area"], default=None)


class ColorDetector(Detector[ObjectDetection]):
    def __init__(
        self,
        color_name: str,
        ball_diameter_cm: float = 6.0,
        focal_px: float = 680.0,
    ) -> None:
        if color_name not in COLOR_CONFIG:
            raise ValueError(f"Couleur non supportée : {color_name}")
        if (not math.isfinite(ball_diameter_cm) or ball_diameter_cm <= 0
                or not math.isfinite(focal_px) or focal_px <= 0):
            raise ValueError("Ball diameter and focal length must be finite and positive")
        self.color_name = color_name
        self.ball_diameter_cm = ball_diameter_cm
        self.focal_px = focal_px

    def detect(self, frame_rgb: NDArray[np.uint8]) -> ObjectDetection | None:
        candidates = detect_colored_balls(frame_rgb, self.color_name, is_rgb=True)
        candidate = select_largest_ball(candidates)
        if candidate is None:
            return None

        x, y, width, height = candidate["bbox"]
        center_x = x + width / 2
        frame_width = frame_rgb.shape[1]
        position = (
            "left" if center_x < frame_width / 3
            else "right" if center_x > 2 * frame_width / 3
            else "center"
        )
        distance = estimate_distance_cm(
            candidate["diameter_px"], self.ball_diameter_cm, self.focal_px,
        )
        return ObjectDetection(
            label=f"{self.color_name}_ball",
            position=position,
            distance_cm=distance,
            distance_visual_cm=distance,
            confidence=0.95,
            diameter_px=candidate["diameter_px"],
            bbox_px=(x, y, width, height),
        )

"""Estimation de distance visuelle pour les objets détectés par la caméra."""

from __future__ import annotations

import math


def estimate_distance_cm(diameter_px: float | None, diameter_cm: float, focal_px: float) -> float | None:
    """d = focal_px * diameter_cm / diameter_px ; None si l'entrée est invalide."""
    try:
        values = (float(diameter_px), float(diameter_cm), float(focal_px))
    except (TypeError, ValueError):
        return None
    if not all(math.isfinite(v) for v in values):
        return None
    diameter, object_cm, focal = values
    if diameter <= 0 or object_cm <= 0 or focal <= 0:
        return None
    return focal * object_cm / diameter

"""YuNet face detection on already enhanced RGB frames; no identity recognition."""

from dataclasses import replace
import logging
import math
from pathlib import Path

import cv2
import numpy as np
from numpy.typing import NDArray

from irobot.config.settings import RuntimeSettings
from .base_detector import Detector
from .distance_estimator import estimate_distance_cm
from irobot.interfaces.perception_contracts import ObjectDetection

logger = logging.getLogger(__name__)


class FaceDetector(Detector[ObjectDetection]):
    def __init__(self, settings: RuntimeSettings, *, focal_px: float | None = None) -> None:
        self._settings = settings
        self._focal_px = settings.camera_focal_px if focal_px is None else focal_px
        if not math.isfinite(self._focal_px) or self._focal_px <= 0:
            raise ValueError("Face focal_px must be finite and positive")
        model = Path(settings.camera_face_model_path)
        if not model.is_file():
            raise FileNotFoundError(f"Modèle YuNet introuvable : {model}")
        self._net = cv2.FaceDetectorYN.create(
            str(model), "", (300, 300),
            score_threshold=settings.camera_face_score_threshold,
            nms_threshold=0.3, top_k=5000,
        )
        self._last_detection: ObjectDetection | None = None
        self._lost_frames = 0
        self._frame_size: tuple[int, int] | None = None
        self._bounding_box: tuple[float, float, float, float] | None = None

    @property
    def bounding_box(self) -> tuple[float, float, float, float] | None:
        return self._bounding_box

    @property
    def is_cached(self) -> bool:
        return self._last_detection is not None and self._lost_frames > 0

    def reset(self) -> None:
        self._last_detection = None
        self._lost_frames = 0
        self._frame_size = None
        self._bounding_box = None

    def detect(self, frame_rgb: NDArray[np.uint8]) -> ObjectDetection | None:
        size = (frame_rgb.shape[1], frame_rgb.shape[0])
        if size != self._frame_size:
            self.reset()
            self._frame_size = size
        try:
            self._net.setInputSize(size)
            _, faces = self._net.detect(cv2.cvtColor(frame_rgb, cv2.COLOR_RGB2BGR))
            detection = self._select_face(faces, size)
        except (cv2.error, ValueError):
            self.reset()
            logger.exception("Échec de la détection YuNet ; mémoire du visage effacée.")
            raise
        if detection is not None:
            self._last_detection = detection
            self._lost_frames = 0
            return replace(detection)
        if self._last_detection is not None and self._lost_frames < self._settings.camera_face_grace_frames:
            self._lost_frames += 1
            logger.debug("Visage mémorisé : image perdue %d/%d.", self._lost_frames,
                         self._settings.camera_face_grace_frames)
            return replace(self._last_detection)
        self._last_detection = None
        self._bounding_box = None
        return None

    def _select_face(
        self, faces: NDArray[np.float32] | None, size: tuple[int, int],
    ) -> ObjectDetection | None:
        if faces is None or faces.size == 0:
            return None
        if faces.ndim != 2 or faces.shape[1] != 15 or not np.isfinite(faces).all():
            raise ValueError("YuNet must return finite rows of 15 values")
        best = None
        for face in faces:
            x, y, width, height = map(float, face[:4])
            confidence = float(face[-1])
            if width <= 0 or height <= 0 or not 0 <= confidence <= 1:
                raise ValueError("YuNet returned invalid face geometry or confidence")
            if confidence < float(np.float32(self._settings.camera_face_score_threshold)):
                continue
            if x + width <= 0 or y + height <= 0 or x >= size[0] or y >= size[1]:
                logger.warning("Visage YuNet hors image ignoré.")
                continue
            if best is None or width > float(best[2]):
                best = face
        if best is None:
            return None
        x, _, width, _ = map(float, best[:4])
        distance = estimate_distance_cm(width, self._settings.camera_face_width_cm, self._focal_px)
        if distance is None or not math.isfinite(distance):
            raise ValueError("Distance visage inexploitable")
        bounding_box = (float(best[0]), float(best[1]), width, float(best[3]))
        self._bounding_box = bounding_box
        center_x = (x + width / 2) * 480 / size[0]
        position = "left" if center_x < 180 else "right" if center_x > 300 else "center"
        return ObjectDetection(
            label="face", position=position, distance_cm=distance,
            distance_visual_cm=distance, confidence=float(best[-1]), diameter_px=width,
            bbox_px=tuple(map(int, bounding_box)),
        )

### FILE: src/irobot/hardware/camera/detector/target_detector.py
"""Dispatch camera frames to the detector for the requested target."""

from __future__ import annotations

import numpy as np
from numpy.typing import NDArray

from irobot.config.settings import MarkerIdentity, RuntimeSettings
from irobot.domain.perception import PerceptionTarget
from irobot.hardware.camera.detector import ColorDetector, FaceDetector, OpenCVMarkerDetector
from irobot.interfaces.perception_contracts import ObjectDetection


class CameraTargetDetector:
    """Dispatch a frame to the detector for the requested semantic target."""

    def __init__(
        self,
        settings: RuntimeSettings,
        *,
        ball_diameter_cm: float,
        focal_px: float,
    ) -> None:
        red = ColorDetector("red", ball_diameter_cm, focal_px)
        yellow = ColorDetector("yellow", ball_diameter_cm, focal_px)
        self._color_detectors = {
            PerceptionTarget.RED_BALL: red,
            PerceptionTarget.YELLOW_BALL: yellow,
        }
        self._marker_detector = OpenCVMarkerDetector(
            settings.human_marker_identity
            or MarkerIdentity("aruco", "DICT_4X4_50", "1"),
            focal_px=focal_px,
            oriented_frame_size_px=settings.camera_oriented_frame_size_px,
        )
        self._settings = settings
        self._focal_px = focal_px
        self._face_detector: FaceDetector | None = None

    def detect(
        self,
        frame_rgb: NDArray[np.uint8],
        target: PerceptionTarget,
    ) -> ObjectDetection | None:
        if target in self._color_detectors:
            return self._color_detectors[target].detect(frame_rgb)
        if target is PerceptionTarget.ANY_BALL:
            return (
                self._color_detectors[PerceptionTarget.RED_BALL].detect(frame_rgb)
                or self._color_detectors[PerceptionTarget.YELLOW_BALL].detect(frame_rgb)
            )
        if target is PerceptionTarget.USER_MARKER:
            return self._marker_detector.detect(frame_rgb).as_object_detection()
        if target is PerceptionTarget.FACE:
            if self._face_detector is None:
                self._face_detector = FaceDetector(self._settings, focal_px=self._focal_px)
            return self._face_detector.detect(frame_rgb)

        lab[:, :, 0] = self._clahe.apply(lab[:, :, 0])
        return cv2.cvtColor(lab, cv2.COLOR_LAB2RGB)

def _attempt_cleanup(errors: list[Exception], resource: Any, method: str, label: str) -> bool:
    try:
        callback = getattr(resource, method, None)
        if not callable(callback):
            return False
        callback()
    except Exception as exc:
        logging.getLogger(__name__).exception("Impossible de nettoyer %s", label)
        errors.append(exc)
        return True
    return True


def close_hardware_runtime(runtime: dict[str, Any]) -> None:
    """Arrête les moteurs et tente de libérer toutes les ressources matérielles."""
    errors: list[Exception] = []
    robot = runtime.get("robot")
    _attempt_cleanup(errors, robot, "stop", "le robot")
    _attempt_cleanup(errors, robot, "cleanup", "le robot")
    _attempt_cleanup(errors, runtime.get("sensor"), "cleanup", "le capteur")
    _attempt_cleanup(errors, runtime.get("camera"), "close", "la caméra")

    if len(errors) == 1:
        raise errors[0]
    if errors:
        raise ExceptionGroup("Échecs pendant la fermeture du runtime hardware", errors)


def build_hardware_runtime(*, settings: RuntimeSettings | None = None) -> dict[str, Any]:
    """Construit les dépendances hardware réelles quand elles sont disponibles (Linux/Pi)."""
    runtime_settings = settings or load_runtime_settings()
    runtime: dict[str, Any] = {"robot": None, "camera": None, "sensor": None}

    try:
        if sys.platform.startswith("linux"):
            try:
                from irobot.hardware.sensors.hc_sr04 import HC_SR04
            except (ModuleNotFoundError, SystemExit) as exc:
                raise RuntimeError("Impossible de charger le pilote HC-SR04") from exc

            sensor = HC_SR04()
            runtime["sensor"] = sensor

            from irobot.hardware.smart_car.smart_car import SmartCar2WD
            from irobot.interfaces.robot_controller import RobotController

            car = SmartCar2WD(

            from irobot.hardware.camera.camera_live_perception_pipeline import CameraLivePerceptionPipeline

            runtime["camera"] = CameraLivePerceptionPipeline(sensor=sensor)
    except BaseException:
        logging.getLogger(__name__).exception("Impossible d'assembler le runtime hardware")
        try:
            close_hardware_runtime(runtime)
        except Exception:

"""Capteur ultrason HC-SR04 sur Raspberry Pi.

Le capteur marche via deux GPIO :
- TRIG : émission d'une impulsion ultrasonore
- ECHO : détection du signal de retour

Le test du projet consiste à mesurer la distance du premier objet situé devant le
capteur.
"""

from __future__ import annotations

import argparse
import os
import threading
import time
from collections.abc import Callable

try:
    from gpiozero import DigitalInputDevice, DigitalOutputDevice
except ModuleNotFoundError as exc:  # pragma: no cover
    raise SystemExit(
        "Ce script nécessite gpiozero sur Raspberry Pi. "
        "Installez le support GPIO puis relancez le programme."
    ) from exc


class HC_SR04:
    """Mesure la distance à l'objet le plus proche devant le capteur."""

    CACHE_TTL_S = 0.05

    def __init__(
        self,
        trigger_pin: int = 17,
        echo_pin: int = 27,
        max_distance_cm: float = 400.0,
        *,
        time_fn: Callable[[], float] = time.monotonic,
        sleep_fn: Callable[[float], None] = time.sleep,
    ) -> None:
        self.trigger = DigitalOutputDevice(trigger_pin)
        self.echo = DigitalInputDevice(echo_pin, pull_up=False)
        self.max_distance_cm = max_distance_cm
        self._time_fn = time_fn
        self._sleep_fn = sleep_fn
        self._read_lock = threading.Lock()
        self._last_distance_cm: float | None = None
        self._last_distance_at: float | None = None
        self._closed = False

    def _trigger_pulse(self) -> None:
        self.trigger.on()
        self._sleep_fn(0.00001)
        self.trigger.off()

    def _wait_for_level(self, expected: int, timeout_s: float = 0.05) -> bool:
        deadline = self._time_fn() + timeout_s
        while self._time_fn() < deadline:
            if self.echo.value == expected:
                return True
            self._sleep_fn(0.00001)
        return False

    def read_distance_cm(self) -> float | None:
        """Retourne une mesure récente en cache, sinon mesure la distance en cm."""
        with self._read_lock:
            now = self._time_fn()
            if (
                self._last_distance_cm is not None
                and self._last_distance_at is not None
                and 0.0 <= now - self._last_distance_at < self.CACHE_TTL_S
            ):
                return self._last_distance_cm

            self._last_distance_cm = None
            self._last_distance_at = None
            self._trigger_pulse()
            if not self._wait_for_level(1):
                return None

            start = self._time_fn()
            if not self._wait_for_level(0):
                return None

            elapsed = self._time_fn() - start
            if elapsed <= 0:
                return None

            distance_cm = (elapsed * 34300.0) / 2.0
            if distance_cm > self.max_distance_cm:
                return None

            self._last_distance_cm = distance_cm
            self._last_distance_at = self._time_fn()
            return distance_cm

    def cleanup(self) -> None:
        if self._closed:
            return
        self._closed = True
        try:
            self.trigger.close()
        finally:
            self.echo.close()


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        description="Mesure la distance du premier objet détecté par un HC-SR04."
    )
    parser.add_argument(
        "--trigger-pin",
        type=int,
        default=17,
        help="GPIO BCM du signal TRIG.",
    )
    parser.add_argument(
        "--echo-pin",
        type=int,
        default=27,
        help="GPIO BCM du signal ECHO.",
    )
    parser.add_argument(
        "--interval",
        type=float,
        default=0.5,
        help="Intervalle entre deux mesures, en secondes.",
    )
    parser.add_argument(
        "--max-distance",
        type=float,
        default=400.0,
        help="Distance maximale mesurable, en cm.",
    )
    parser.add_argument(
        "--once",
        action="store_true",
        help="Mesure une seule valeur puis quitte.",
    )
    return parser


def main() -> None:
    args = build_parser().parse_args()
    sensor = HC_SR04(
        trigger_pin=args.trigger_pin,
        echo_pin=args.echo_pin,
        max_distance_cm=args.max_distance,
    )

    try:
        print(f"Mesure HC-SR04 via TRIG={args.trigger_pin} / ECHO={args.echo_pin}")
        print("Objectif : afficher la distance du premier objet devant le capteur.")

        while True:
            distance = sensor.read_distance_cm()
            if distance is None:
                print("Aucun objet détecté dans la plage mesurée.")
            else:
                print(f"Distance détectée : {distance:.2f} cm")

            if args.once:
                break
            time.sleep(args.interval)
    except KeyboardInterrupt:
        print("\nInterruption clavier : arrêt du capteur.")
    finally:
        sensor.cleanup()
        if args.once:
            print("Fermeture propre du capteur...")

if __name__ == "__main__":
    main()

def build_parser() -> argparse.ArgumentParser:
    """Crée le parser de ligne de commande."""
    settings = load_runtime_settings()
    parser = argparse.ArgumentParser(
        description="Contrôle d'une voiture Smart Car 2WD sur Raspberry Pi via un pont H-bridge L298N."
    )
    parser.add_argument(

        choices=["forward", "backward", "left", "right", "stop"],

    parser.add_argument(

    parser.add_argument(

    )
    parser.add_argument(

        type=float,
        default=settings.obstacle_threshold_cm,
        help="Arrête la voiture si le capteur HC-SR04 détecte un objet à moins de X cm.",
    )
    parser.add_argument(
        "--obstacle-check-interval",
        type=float,
        default=0.2,
        help="Intervalle de vérification du capteur en secondes pendant un déplacement.",
    )
    return parser


def run_command(args: argparse.Namespace) -> None:
    """Exécute une commande via le contrôleur et libère les ressources."""
    if args.duration < -1:
        raise ValueError("La durée doit être >= -1. (-1 = indéfinie)")
    if not 0.0 <= args.speed <= 1.0:
        raise ValueError("La vitesse doit être comprise entre 0.0 et 1.0.")
    if args.obstacle_threshold_cm < 0:
        raise ValueError("Le seuil d'obstacle doit être positif.")

    with ExitStack() as cleanup:
        sensor = HC_SR04() if HC_SR04 is not None else None
        if sensor is not None:
            cleanup.callback(sensor.cleanup)

        car = SmartCar2WD(drive_speed=args.speed)
        cleanup.callback(car.cleanup)
        controller = RobotController(
            car,
            distance_sensor=sensor,
            obstacle_threshold_cm=args.obstacle_threshold_cm,
        )
        command = getattr(controller, args.command)

        if args.command == "stop":
            command()
            print("Commande 'stop' exécutée.")
            return

        command()
        check_obstacles = args.command != "backward"
        if args.duration == -1:
            print(
                f"Commande '{args.command}' en cours indéfiniment "
                f"(vitesse {args.speed:.2f}, seuil obstacle {args.obstacle_threshold_cm:.1f} cm). "
                "Appuie sur Ctrl+C pour stopper."
            )
            while True:
                if check_obstacles and controller.check_obstacle():
                    _stop_for_obstacle(controller, args.obstacle_threshold_cm)
                    return
                time.sleep(args.obstacle_check_interval)

        if args.duration > 0:
            deadline = time.monotonic() + args.duration
            while time.monotonic() < deadline:
                if check_obstacles and controller.check_obstacle():
                    _stop_for_obstacle(controller, args.obstacle_threshold_cm)
                    return
                time.sleep(args.obstacle_check_interval)

            controller.stop()
            print(
                f"Commande '{args.command}' exécutée pendant {args.duration}s à vitesse {args.speed:.2f}."
            )
            return

        controller.stop()
        print(f"Commande '{args.command}' arrêtée immédiatement.")


def _stop_for_obstacle(controller: RobotController, threshold_cm: float) -> None:
    controller.stop()
    print(f"Obstacle détecté à moins de {threshold_cm:.1f} cm : arrêt de la voiture.")


def _stop_lgpio_notify_thread() -> None:
    """Arrête le thread daemon de lgpio à la finalisation de Python."""
    try:
        import lgpio
        from gpiozero import Device
    except ImportError:  # pragma: no cover - hors Raspberry Pi
        return

    thread = getattr(lgpio, "_notify_thread", None)
    if thread is None:
        return
    try:
        thread.stop()
        if Device.pin_factory is not None:
            Device.pin_factory.close()
        lgpio._notify_close(thread._notify)
        thread.join(1.0)
    except Exception:
        pass


def main() -> None:
    """Point d'entrée du script."""
    parser = build_parser()
    args = parser.parse_args()

    try:
        run_command(args)
    except ValueError as exc:
        parser.error(str(exc))
    except KeyboardInterrupt:
        print("\nInterruption clavier : arrêt des moteurs.")
    finally:
        _stop_lgpio_notify_thread()

"""Pilote matériel d'une voiture 2 roues via un module L298N.

Notes de câblage :
- GPIO 5, 6, 16, 24 sont utilisés pour les entrées IN1, IN2, IN3, IN4
- GPIO 12 et 13 pilotent ENA et ENB (PWM pour la vitesse)
- le GND du Raspberry Pi doit être relié au GND du module L298N
"""

from contextlib import ExitStack

try:
    from gpiozero import DigitalOutputDevice, PWMOutputDevice
except ModuleNotFoundError:  # pragma: no cover - relevant on non-Pi hosts only
    DigitalOutputDevice = None
    PWMOutputDevice = None


class SmartCar2WD:
    """Contrôle une voiture 2 roues motrices (2WD) avec un L298N.

    Les broches GPIO sont celles du mode BCM (Broadcom), pas les numéros de broches
    physiques du connecteur 40 broches.
    """

    DEFAULT_PINS = {
        "in1": 5,
        "in2": 6,
        "in3": 16,
        "in4": 24,
        "ena": 12,
        "enb": 13,
    }

    def __init__(
        self,
        in1=DEFAULT_PINS["in1"],
        in2=DEFAULT_PINS["in2"],
        in3=DEFAULT_PINS["in3"],
        in4=DEFAULT_PINS["in4"],
        ena=DEFAULT_PINS["ena"],
        enb=DEFAULT_PINS["enb"],
        drive_speed: float = 0.5,
    ):
        self.pins = {
            "in1": DigitalOutputDevice(in1),
            "in2": DigitalOutputDevice(in2),
            "in3": DigitalOutputDevice(in3),
            "in4": DigitalOutputDevice(in4),
        }
        self.ena = PWMOutputDevice(ena, frequency=100)
        self.enb = PWMOutputDevice(enb, frequency=100)
        self._closed = False

        self.speed = self.clamp_speed(drive_speed)
        self.stop()

    @staticmethod
    def clamp_speed(speed: float) -> float:
        """Limite la vitesse à l'intervalle [0.0, 1.0]."""
        return max(0.0, min(1.0, speed))

    def set_speed(self, speed: float) -> None:
        """Applique une vitesse globale sur les deux moteurs."""
        self.speed = self.clamp_speed(speed)
        self.ena.value = self.speed
        self.enb.value = self.speed

    def _set_motor_a(self, forward: bool) -> None:
        """Configure le moteur A (roue gauche ou droite selon le schéma)."""
        self.pins["in1"].value = 1 if forward else 0
        self.pins["in2"].value = 0 if forward else 1
        self.ena.value = self.speed

    def _set_motor_b(self, forward: bool) -> None:
        """Configure le moteur B."""
        self.pins["in3"].value = 1 if forward else 0
        self.pins["in4"].value = 0 if forward else 1
        self.enb.value = self.speed

    def forward(self) -> None:
        """Avance en ligne droite."""
        self._set_motor_a(True)
        self._set_motor_b(True)

    def backward(self) -> None:
        """Recule en ligne droite."""
        self._set_motor_a(False)
        self._set_motor_b(False)

    def left(self) -> None:
        """Tourne vers la gauche.

        Le moteur A est inversé tandis que le moteur B avance.
        """
        self._set_motor_a(False)
        self._set_motor_b(True)

    def right(self) -> None:
        """Tourne vers la droite.

        Le moteur A avance tandis que le moteur B est inversé.
        """
        self._set_motor_a(True)
        self._set_motor_b(False)

    def stop(self) -> None:
        """Arrête les moteurs et met la vitesse PWM à zéro."""
        for pin in self.pins.values():
            pin.off()
        self.ena.value = 0
        self.enb.value = 0

    def cleanup(self) -> None:
        """Arrête le véhicule et libère ses GPIO."""
        if self._closed:
            return
        self._closed = True
        try:
            self.stop()
        finally:
            with ExitStack() as cleanup:
                for pin in self.pins.values():
                    cleanup.callback(pin.close)
                cleanup.callback(self.ena.close)
                cleanup.callback(self.enb.close)

    ) -> None:

        self.controller = LanguageController(

        )

        self.warmup: LLMWarmup | None = None
        self.wait_for_warmup = False

    def _ask_plan(self, text: str) -> tuple[list[Intent], dict[str, Any]]:
        return self.llm_parser.parse_plan_with_trace(text)

    def _emergency_stop_requested(self) -> bool:
        if self._emergency_event.is_set():
            return True
        detector = getattr(self.robot, "is_emergency_stop_requested", None)
        if not callable(detector):
            return False
        try:
            return bool(detector())
        except Exception:
            logging.getLogger(__name__).exception("Impossible de lire l'arret d'urgence")
            return True

    def safe_stop(self) -> None:
        stop = getattr(self.robot, "stop", None)
        if callable(stop):
            try:
                stop()
            except Exception:
                logging.getLogger(__name__).exception("Impossible d'arreter le robot")

    def _await_warmup(self) -> str:
        if self.warmup is None:
            return "skipped"
        if not self.wait_for_warmup:
            return self.warmup.status
        wait_seconds = llm_warmup_wait_seconds()
        if self.warmup.status == "running":
            print(f"[LLM] Préchauffage du modèle en cours (attente max {wait_seconds:g}s)...", flush=True)
        status = self.warmup.wait(wait_seconds)
        if status == "timeout":
            print(f"[LLM] Préchauffage non terminé après {wait_seconds:g}s, envoi de la commande quand même.", flush=True)
        elif status == "failed":
            print("[LLM] Préchauffage en échec, envoi de la commande quand même.", flush=True)
        return status

    def _trace_path(self) -> Path:
        stamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S%fZ")
        return self.trace_dir / f"runtime_trace_{stamp}.jsonl"

    def _write_trace_event(self, trace_path: Path, event: dict[str, Any]) -> None:
        payload = {"timestamp": datetime.now(timezone.utc).isoformat(), **event}
        with trace_path.open("a", encoding="utf-8") as fh:
            fh.write(json.dumps(payload, ensure_ascii=False) + "\n")

    def _trace_mission_event(self, trace_path: Path, event: dict[str, Any]) -> None:
        event_type = event["event"]
        intent = event["intent"]
        if event_type == "step_result":
            result = event["result"]
            self._write_trace_event(trace_path, {
                "event": event_type,
                "attempt": event["attempt"],
                "intent": asdict(intent),
                "status": result.get("status"),
                "duration_seconds": round(event["duration_seconds"], 4),
                "results": [
                    {"status": item.status, "message": item.message, "data": item.data}
                    for item in result.get("results") or []
                ],
            })
        elif event_type in {"emergency_stop", "obstacle_stop"}:
            self._write_trace_event(trace_path, {
                "event": event_type,
                "reason": event["reason"],
                "intent": asdict(intent),
            })
        else:
            self._write_trace_event(trace_path, {
                "event": event_type,
                "intent": asdict(intent),
            })

    def handle_text(self, text: str) -> dict[str, Any]:
        """Parse, validates, plans, and executes a complete text command."""
        trace_path = self._trace_path()
        started_at = time.time()
        self._write_trace_event(trace_path, {"event": "input", "text": text})
        mission = self.mission_orchestrator.state_machine
        mission.reset()
        intents, plan_trace, warmup_status = self._collect_plan(text)
        if self._emergency_stop_requested():
            mission.abort("emergency stop")
            return {
                "input": text, "status": "aborted", "reason": "emergency stop",
                "steps": [], "trace_path": str(trace_path), "mission": mission.as_dict(),
            }
        refusal = self._plan_refusal(intents, plan_trace)
        if refusal is not None:
            return {"status": "rejected", "reason": refusal, "steps": []}
        self._trace_raw_plan(trace_path, intents, plan_trace, warmup_status)
        if plan_trace.get("source") in UNSUPPORTED_PLAN_SOURCES:
            return self._unsupported_plan(text, trace_path, plan_trace)
        planned_steps, validation_errors = self._prepare_plan(text, intents, trace_path)
        if validation_errors:
            mission.fail("invalid plan step")
            return {
                "input": text, "status": "rejected", "reason": "invalid plan step",
                "validation_errors": validation_errors, "trace_path": str(trace_path),
                "source": plan_trace.get("source"), "mission": mission.as_dict(),
            }
        if not planned_steps:
            mission.fail("empty plan")
            return {
                "status": "rejected", "reason": "empty plan", "input": text,
                "trace_path": str(trace_path), "source": plan_trace.get("source"),
                "mission": mission.as_dict(),
            }
        return self._execute_plan(text, planned_steps, validation_errors, plan_trace, trace_path, started_at)

    def _execute_plan(
        self, text: str, planned_steps: list[SkillPlanStep],
        validation_errors: list[dict[str, Any]], plan_trace: dict[str, Any],
        trace_path: Path, started_at: float,
    ) -> dict[str, Any]:
        try:
            steps, status = self.mission_orchestrator.execute(
                planned_steps, self.controller.handle_planned_step,
                stop_requested=self._emergency_stop_requested,
                on_event=lambda event: self._trace_mission_event(trace_path, event),
            )
        finally:
            self.safe_stop()
        duration = round(time.time() - started_at, 4)
        self._write_trace_event(trace_path, {
            "event": "mission_result", "status": status,
            "duration_seconds": duration, "steps_count": len(steps),
        })
        return {
            "input": text, "status": status, "steps": steps,
            "validation_errors": validation_errors, "source": plan_trace.get("source"),
            "trace_path": str(trace_path), "duration_seconds": duration,
            "mission": self.mission_orchestrator.state_machine.as_dict(),
        }

    def _collect_plan(self, text: str) -> tuple[list[Intent], dict[str, Any], str]:
        warmup_status = self.warmup.status if self.warmup is not None else "skipped"
        if is_pure_stop(text):
            # Sécurité : arrêt pur traité localement, sans LLM ni normalisation des nombres.
            intents = [Intent(name="stop", raw_text=text, confidence=1.0)]
            plan_trace = {"source": "local_stop", "llm_content": None, "llm_json": None, "normalized_text": None}
        elif self.use_llm:
            warmup_status = self._await_warmup()
            intents, plan_trace = self._ask_plan(text)
        else:
            intents = self.parser.parse_plan(text)
            plan_trace = {
                "source": "local_parser",
                "llm_content": None,
                "llm_json": None,
                "normalized_text": preprocess(text),
            }
        return intents, plan_trace, warmup_status

    def _plan_refusal(self, intents: list[Intent], plan_trace: dict[str, Any]) -> str | None:
        if plan_trace.get("source") == "llm_follow_refused":
            return plan_trace["guard_reason"]
        if any(
            follow_request_refusal(it.raw_text)
            or contains_unsupported_follow_step({
                "intent": it.name, "target": it.target, "parameters": it.parameters,
            }) for it in intents
        ):
            return "unsupported_follow_target"
        return None

    def _trace_raw_plan(
        self, trace_path: Path, intents: list[Intent],
        plan_trace: dict[str, Any], warmup_status: str,
    ) -> None:
        self._write_trace_event(
            trace_path,
            {
                "event": "plan_raw",
                "source": plan_trace.get("source"),
                "normalized_text": plan_trace.get("normalized_text"),
                "intents": [asdict(it) for it in intents],
                "llm_json": plan_trace.get("llm_json"),
                "llm_content": plan_trace.get("llm_content"),
                "error": plan_trace.get("error"),
                "error_type": plan_trace.get("error_type"),
                "guard_notes": plan_trace.get("guard_notes"),
                "guard_reason": plan_trace.get("guard_reason"),
                "llm_latency_seconds": plan_trace.get("llm_latency_seconds"),
                "llm_response_format": plan_trace.get("llm_response_format"),
                "llm_attempts": plan_trace.get("llm_attempts"),
                "llm_prompt_tokens": plan_trace.get("llm_prompt_tokens"),
                "llm_completion_tokens": plan_trace.get("llm_completion_tokens"),
                "llm_timings": plan_trace.get("llm_timings"),
                "warmup_status": warmup_status,
            },
        )

    def _unsupported_plan(self, text: str, trace_path: Path, plan_trace: dict[str, Any]) -> dict[str, Any]:
        mission = self.mission_orchestrator.state_machine
        mission.fail("unsupported command")
        self._write_trace_event(trace_path, {
            "event": "plan_unsupported", "text": text,
            "normalized_text": plan_trace.get("normalized_text"),
            "llm_content": plan_trace.get("llm_content"),
            "guard_reason": plan_trace.get("guard_reason"),
            "guard_notes": plan_trace.get("guard_notes"),
        })
        return {
            "input": text, "status": "rejected", "reason": "unsupported command",
            "guard_reason": plan_trace.get("guard_reason"), "steps": [],
            "trace_path": str(trace_path), "source": plan_trace.get("source"),
            "mission": mission.as_dict(),
        }

    def _prepare_plan(
        self, text: str, intents: list[Intent], trace_path: Path,
    ) -> tuple[list[SkillPlanStep], list[dict[str, Any]]]:
        intents = self.plan_validator.normalize(intents)
        intents = self.planner.expand_intents(intents)

        if len(intents) > int(self.settings.max_plan_steps):
            intents = intents[: int(self.settings.max_plan_steps)]
            intents.append(Intent(name="stop", raw_text=text, confidence=1.0))

        validated_intents, validation_errors = self.plan_validator.validate(intents)
        planned_steps = self.planner.plan_intents(validated_intents)
        self._write_trace_event(
            trace_path,
            {
                "event": "plan_validated",
                "intents": [asdict(it) for it in validated_intents],
                "validation_errors": validation_errors,
            },
        )
        self._write_trace_event(
            trace_path,
            {
                "event": "plan_compiled",
                "steps": [
                    {
                        "intent": asdict(step.intent),
                        "skills": [skill.name for skill in step.skills],
                    }
                    for step in planned_steps
                ],
            },
        )

        return planned_steps, validation_errors

def _llm_confidence(value: Any) -> float:
    if value is None or isinstance(value, bool):
        return LLM_DEFAULT_CONFIDENCE
    try:
        confidence = float(value)
    except (TypeError, ValueError):
        return LLM_DEFAULT_CONFIDENCE
    if not math.isfinite(confidence):
        return LLM_DEFAULT_CONFIDENCE
    return max(0.0, min(1.0, confidence))


def _intent_from_llm_step(step: Any, raw_text: str) -> Intent | None:
    """Reconstruit un Intent depuis un objet JSON du LLM ; None si l'objet est malformé."""
    if not isinstance(step, dict):
        return None
    parameters = step.get("parameters")
    if parameters is None:
        parameters = {}
    if not isinstance(parameters, dict):
        return None
    target = step.get("target")
    return Intent(
        name=str(step.get("intent") or step.get("name") or "unknown").strip(),
        target=str(target) if target is not None else None,
        parameters=dict(parameters),
        confidence=_llm_confidence(step.get("confidence")),
        raw_text=raw_text,
    )


def _normalize_llm_step(step: Any) -> dict[str, Any] | None:
    """Étape LLM tolérante (target absent -> None, parameters absent -> {}) ; None si malformée."""
    if not isinstance(step, dict):
        return None
    parameters = step.get("parameters")
    if parameters is None:
        parameters = {}
    if not isinstance(parameters, dict):
        return None
    name = str(step.get("intent") or step.get("name") or "unknown").strip().lower()
    return {
        **step,
        "intent": PLAN_INTENT_SYNONYMS.get(name, name),
        "target": step.get("target"),
        "parameters": dict(parameters),
    }


def ask_llm_for_intent_plan_with_trace(
    text: str, *, post: LLMPost = _post_llm_chat,
) -> tuple[list[Intent], dict[str, Any]]:
    """Demande au LLM un plan ordonné d'intentions et retourne des métadonnées de trace.

    - JSON valide avec `steps: []` -> ([], source "llm_unsupported") : commande hors périmètre ;
    - plan non justifié par la phrase prétraitée (`grounding_check`) -> ([], source "llm_ungrounded") ;
    - timeout, erreur HTTP, JSON invalide, `steps` absent/non liste ou aucune étape exploitable
      -> repli sur TextIntentParser (texte d'origine).
    """
    refusal = follow_request_refusal(text)
    if refusal:
        return [], {"source": "llm_follow_refused", "guard_reason": refusal}
    normalized_text = preprocess(text)
    settings = load_llm_settings()
    payload = _llm_payload(SYSTEM_PROMPT_PLAN, normalized_text, settings.max_tokens_plan, settings.model)
    meta: dict[str, Any] = {}

    def _fallback(console_reason: str, **trace: Any) -> tuple[list[Intent], dict[str, Any]]:
        print(f"[LLM] {console_reason}, repli sur le parseur local", flush=True)
        return TextIntentParser().parse_plan(text), {
            "source": "local_parser_fast_fallback",
            "llm_content": None,
            "llm_json": None,
            "normalized_text": normalized_text,
            **meta,
            **trace,
        }

    try:
        content = post(
            payload,
            (settings.connect_timeout_seconds, settings.read_timeout_seconds),
            _select_formats(PLAN_RESPONSE_FORMATS),
            meta,
        )
    except Exception as exc:
        return _fallback(
            _llm_error_reason(exc, settings.read_timeout_seconds),
            error=f"{type(exc).__name__}: {exc}",
            error_type=type(exc).__name__,
        )

    return _interpret_plan_response(text, normalized_text, content, meta, _fallback)


def _interpret_plan_response(
    text: str, normalized_text: str, content: str, meta: dict[str, Any],
    fallback: Callable[..., tuple[list[Intent], dict[str, Any]]],
) -> tuple[list[Intent], dict[str, Any]]:
    parsed = _parse_llm_json_payload(content)
    if not parsed:
        return fallback("JSON invalide", llm_content=content, reason="invalid json")
    if contains_unsupported_follow_step(parsed):
        return [], {"source": "llm_follow_refused", "guard_reason": "unsupported_follow_target"}

    raw_steps = parsed.get("steps")
    if not isinstance(raw_steps, list):
        return fallback("réponse sans « steps »", llm_content=content, llm_json=parsed, reason="missing steps")

    trace: dict[str, Any] = {"llm_content": content, "llm_json": parsed, "normalized_text": normalized_text, **meta}
    if not raw_steps:
        return [], {"source": "llm_unsupported", **trace}

    steps: list[dict[str, Any]] = []
    skipped: list[int] = []
    for index, step in enumerate(raw_steps):
        normalized_step = _normalize_llm_step(step)
        if normalized_step is None:
            skipped.append(index)
            continue
        steps.append(normalized_step)

    if not steps:
        return fallback(
            "aucune étape exploitable", llm_content=content, llm_json=parsed, reason="no usable step",
            skipped_steps=skipped,
        )

    # Le LLM propose, le lexique dispose : contrôle sur le texte réellement envoyé au LLM.
    kept, guard_reason, guard_notes = grounding_check(normalized_text, steps)
    trace["guard_notes"] = guard_notes
    trace["skipped_steps"] = skipped
    if guard_reason or not kept:
        return [], {"source": "llm_ungrounded", **trace, "guard_reason": guard_reason or "no grounded step"}

    intents = [intent for step in kept if (intent := _intent_from_llm_step(step, text)) is not None]
    return intents, {"source": "llm_plan", **trace}


class LLMIntentParser:
    """LLM-backed parser implementing the shared intent-plan contract."""

    def __init__(
        self,
        parse_with_trace: Callable[[str], tuple[list[Intent], dict[str, Any]]] | None = None,
    ):
        self._parse_with_trace = parse_with_trace or ask_llm_for_intent_plan_with_trace

    def parse_plan(self, text: str) -> list[Intent]:
        intents, _ = self.parse_plan_with_trace(text)
        return intents

    def parse_plan_with_trace(self, text: str) -> tuple[list[Intent], dict[str, Any]]:

def warm_up_llm(timeout: tuple[float, float] | None = None) -> bool:
    """Requête minimale avec le prompt système du plan pour remplir le cache de prompt du serveur."""
    settings = load_llm_settings()
    payload = _llm_payload(SYSTEM_PROMPT_PLAN, "stop", 1, settings.model)
    try:
        _post_llm_chat(
            payload,
            timeout or (settings.connect_timeout_seconds, LLM_WARMUP_READ_TIMEOUT_SECONDS),
            _select_formats(PLAN_RESPONSE_FORMATS),

        return True
    except Exception as exc:
        print(f"[LLM] Echec du prechauffage : {exc}", flush=True)
        return False


class LLMWarmup:
    """Préchauffage en thread daemon ; `wait()` le rend bloquant (mode une seule commande).

    Statuts : skipped (non lancé) | running | done | failed | timeout (attente expirée, définitif).
    """

    def __init__(self) -> None:
        self._thread: threading.Thread | None = None
        self._result: bool | None = None
        self._timed_out = False

    def start(self) -> "LLMWarmup":
        self._thread = threading.Thread(target=self._run, name="llm-warmup", daemon=True)
        self._thread.start()
        return self

    def _run(self) -> None:
        self._result = warm_up_llm()

    def wait(self, timeout: float) -> str:
        if self._thread is not None and self._thread.is_alive():
            self._thread.join(timeout)
            if self._thread.is_alive():
                self._timed_out = True
        return self.status

    @property
    def status(self) -> str:
        if self._thread is None:
            return "skipped"
        if self._timed_out:
            return "timeout"
        if self._thread.is_alive():
            return "running"
        return "done" if self._result else "failed"


def warm_up_llm_async() -> threading.Thread:
    """Lance le préchauffage en arrière-plan (thread daemon) sans bloquer le démarrage."""
    thread = threading.Thread(target=warm_up_llm, name="llm-warmup", daemon=True)
    thread.start()
    return thread

"""Normalize and validate parsed intents before compiling a skill plan."""

from __future__ import annotations

import math
from dataclasses import asdict
from typing import Any

from irobot.config.settings import RuntimeSettings
from irobot.domain.skills.primitives.base_motion_primitive import normalize_distance_unit
from irobot.domain.intent import Intent
from irobot.domain.intent_validator import IntentValidator
from irobot.interfaces.plan_guards import (
    contains_unsupported_follow_step, follow_request_refusal, TARGET_COLOR_SYNONYMS,
)
from irobot.interfaces.voice_constants import (
    DEFAULT_MOVE_DURATION_SEC,
    DEFAULT_TURN_ANGLE_DEG,
    PLAN_INTENT_SYNONYMS,
    OBJECT_INTENTS,
    DEFAULT_STAND_OFF_CM,
    DEFAULT_FOLLOW_DURATION_SEC,
    MIN_OBJECT_DURATION_SEC,
    MIN_SEARCH_TIMEOUT_SEC,
    MAX_SEARCH_TIMEOUT_SEC,
)


class IntentPlanValidator:
    def __init__(self, settings: RuntimeSettings) -> None:
        self.settings = settings
        self.validator = IntentValidator()

    def validate(self, intents: list[Intent]) -> tuple[list[Intent], list[dict[str, Any]]]:
        sanitized: list[Intent] = []
        errors: list[dict[str, Any]] = []
        for index, intent in enumerate(intents):
            if follow_request_refusal(intent.raw_text) or not self.validator.validate(intent):
                errors.append(
                    {
                        "index": index,
                        "reason": "intent not allowed",
                        "intent": asdict(intent),
                    }
                )
                continue
            if intent.name in {"move_forward", "move_backward", "turn_left", "turn_right"}:
                intent = self._sanitize_primitive_intent(intent)
            elif intent.name in OBJECT_INTENTS:
                intent = self._sanitize_object_intent(intent)
            sanitized.append(intent)

        return sanitized, errors

    def normalize(self, intents: list[Intent]) -> list[Intent]:
        """Normalise les noms d'intents issus du LLM pour limiter les `unknown`."""
        synonyms = PLAN_INTENT_SYNONYMS
        normalized: list[Intent] = []
        for intent in intents:
            name = intent.name.strip().lower()
            mapped = synonyms.get(name, name)
            target = intent.target
            parameters = dict(intent.parameters)
            if mapped in OBJECT_INTENTS:
                target, parameters = self._normalize_object_target(mapped, target, parameters)
            normalized.append(
                Intent(
                    name=mapped,
                    target=target,
                    parameters=parameters,
                    confidence=intent.confidence,
                    raw_text=intent.raw_text,
                )
            )
        return normalized

    @staticmethod
    def _color_from_label(value: Any) -> str | None:
        if value is None:
            return None
        return TARGET_COLOR_SYNONYMS.get(str(value).strip().lower())

    def _normalize_object_target(
        self, name: str, target: str | None, parameters: dict[str, Any]
    ) -> tuple[str | None, dict[str, Any]]:
        """Ramène les couleurs de cible (jaune, balle rouge...) à `yellow`/`red`."""
        if name == "follow_object" and contains_unsupported_follow_step({
            "intent": name, "target": target, "parameters": parameters,
        }):
            return target, parameters
        color = self._color_from_label(target)
        if color is None:
            color = self._color_from_label(parameters.get("target_object"))
        if color is not None:
            target = color
            parameters["target_object"] = color
        elif target is None and parameters.get("target_object") is not None and name == "follow_object":
            target = str(parameters["target_object"])
        return target, parameters

    def _to_float(self, value: Any, default: float | None = None) -> float | None:
        try:
            number = float(value)
        except (TypeError, ValueError):
            return default
        return number if math.isfinite(number) else default

    def _sanitize_object_intent(self, intent: Intent) -> Intent:
        """Borne les paramètres de recherche/approche/suivi ; valeurs invalides -> défauts."""
        params = dict(intent.parameters)
        settings = self.settings

        if "stand_off_cm" in params:
            value = self._to_float(params["stand_off_cm"], DEFAULT_STAND_OFF_CM)
            params["stand_off_cm"] = max(
                float(settings.min_stand_off_cm), min(value, float(settings.max_stand_off_cm))
            )
        if "duration_s" in params:
            value = self._to_float(params["duration_s"], DEFAULT_FOLLOW_DURATION_SEC)
            params["duration_s"] = max(MIN_OBJECT_DURATION_SEC, min(value, float(settings.max_follow_duration_sec)))
        if "search_timeout_s" in params:
            value = self._to_float(params["search_timeout_s"], float(settings.search_timeout_seconds))
            params["search_timeout_s"] = max(MIN_SEARCH_TIMEOUT_SEC, min(value, MAX_SEARCH_TIMEOUT_SEC))

        return Intent(
            name=intent.name,
            target=intent.target,
            parameters=params,
            confidence=intent.confidence,
            raw_text=intent.raw_text,
        )

    def _sanitize_primitive_intent(self, intent: Intent) -> Intent:
        """Normalise et borne les paramètres de primitives pour un plan sûr."""
        params = dict(intent.parameters)
        if intent.name in {"move_forward", "move_backward"}:
            distance = self._to_float(params.get("distance"), None)
            unit = normalize_distance_unit(params.get("unit"))
            duration = self._to_float(params.get("duration"), None)

            if distance is None and duration is None:
                duration = DEFAULT_MOVE_DURATION_SEC

            if distance is not None:
                if unit == "m":
                    distance = max(0.01, min(distance, self.settings.max_distance_cm / 100.0))
                else:
                    distance = max(0.1, min(distance, self.settings.max_distance_cm))
                params["distance"] = distance
                params["unit"] = unit

            if duration is not None:
                params["duration"] = max(0.1, min(duration, self.settings.max_duration_sec))

        if intent.name in {"turn_left", "turn_right"}:
            angle = self._to_float(params.get("angle"), None)
            duration = self._to_float(params.get("duration"), None)
            if angle is None and duration is None and not params.get("full_rotation"):
                angle = DEFAULT_TURN_ANGLE_DEG
            if angle is not None:
                params["angle"] = max(1.0, min(angle, self.settings.max_angle_deg))
                params["unit"] = "degrees"
            if duration is not None:
                params["duration"] = max(0.1, min(duration, self.settings.max_duration_sec))

        return Intent(
            name=intent.name,
            target=intent.target,
            parameters=params,
            confidence=intent.confidence,
            raw_text=intent.raw_text,
        )

"""Parsing des intentions issues d'une commande vocale ou textuelle.

Le rôle de ce module est de convertir une phrase naturelle en une intention
métier structurée, puis de laisser l'orchestrateur sélectionner les skills
compatibles. Le LLM n'est pas autorisé à piloter directement les GPIO ; il
produit un `Intent` validé par le système.
"""

from __future__ import annotations

import re
from typing import Any

from irobot.domain.intent import Intent
from irobot.interfaces.plan_guards import fold, follow_request_refusal
from irobot.interfaces.text_normalizer import fr_numbers_to_digits
_DECIMAL_RE = re.compile(r"(\d+)[.,](\d+)")
_DECIMAL_MARK = "\ue000"


def _replace_french_numbers(text: str) -> str:
    """Nombres français en lettres -> chiffres (normaliseur partagé avec le LLM), décimales préservées.

    Les conversions d'unités de `preprocess` ne sont pas appliquées : les extracteurs
    ci-dessous gèrent déjà minutes et mètres (y compris décimaux).
    """
    protected = _DECIMAL_RE.sub(rf"\1{_DECIMAL_MARK}\2", text)
    return fr_numbers_to_digits(protected).replace(_DECIMAL_MARK, ".")


def _extract_color(text: str) -> str | None:
    lowered = text.lower()
    if re.search(r"\b(?:jaune|yellow)\b", lowered):
        return "yellow"
    if re.search(r"\b(?:rouge|red)\b", lowered):
        return "red"
    return None


def _extract_duration(text: str) -> float | None:
    """Extrait une durée en secondes (secondes ou minutes) depuis une phrase libre."""
    lowered = _replace_french_numbers(text.lower())
    total = 0.0
    found = False

    minutes = re.search(r"\b(\d+(?:[.,]\d+)?)\s*(?:minutes?|mins?)\b", lowered)
    if minutes:
        total += float(minutes.group(1).replace(",", ".")) * 60.0
        found = True

    patterns = [
        r"(?:pendant|pour|durant)\s*(\d+(?:[.,]\d+)?)\s*(?:seconde|secondes|sec|secs|s)\b",
        r"(?:for|during)\s*(\d+(?:[.,]\d+)?)\s*(?:second|seconds|sec|secs|s)\b",
        r"\b(\d+(?:[.,]\d+)?)\s*(?:s|sec|second|seconde|seconds|secondes)\b",
    ]
    for pattern in patterns:
        match = re.search(pattern, lowered)
        if match:
            total += float(match.group(1).replace(",", "."))
            found = True
            break

    return max(0.1, total) if found else None


def _extract_distance(text: str) -> tuple[float, str] | None:
    """Extrait une distance (cm ou m) depuis une phrase libre."""
    patterns = [
        r"\b(\d+(?:[.,]\d+)?)\s*(cm|centim[eè]tre|centim[eè]tres)\b",
        r"\b(\d+(?:[.,]\d+)?)\s*(m|m[eè]tre|m[eè]tres)\b",
    ]
    lowered = _replace_french_numbers(text.lower())
    for pattern in patterns:
        match = re.search(pattern, lowered)
        if not match:
            continue
        value = float(match.group(1).replace(",", "."))
        unit = match.group(2)
        if unit.startswith("cm") or unit.startswith("centim"):
            return (max(0.1, value), "cm")
        return (max(0.01, value), "m")
    return None


def _extract_angle(text: str) -> float | None:
    """Extrait un angle en degrés (« de 45 degrés », « quatre vingt dix degrés »)."""
    lowered = _replace_french_numbers(text.lower())
    match = re.search(r"\b(\d+(?:[.,]\d+)?)\s*(?:°|degr[eé]s?|deg)(?:\b|$)", lowered)
    if match:
        return float(match.group(1).replace(",", "."))
    return None


def _motion_parameters(text: str) -> dict[str, Any]:
    """Avance/recul : durée explicite, sinon distance, sinon 1 s par défaut."""
    duration = _extract_duration(text)
    if duration is not None:
        return {"duration": duration}
    distance = _extract_distance(text)
    if distance is not None:
        return {"distance": distance[0], "unit": distance[1]}
    return {"duration": 1.0}


def _turn_parameters(text: str) -> dict[str, Any]:
    """Rotation : durée explicite, sinon angle, sinon 90°."""
    duration = _extract_duration(text)
    if duration:
        return {"duration": duration}
    angle = _extract_angle(text)
    return {"angle": angle if angle is not None else 90, "unit": "degrees"}


def _extract_stand_off_cm(text: str) -> float | None:
    """Extrait une distance d'approche (« à 1 m », « à 50 cm ») convertie en cm."""
    lowered = _replace_french_numbers(text.lower())
    match = re.search(
        r"(?:\bà|\ba|\bde|\bat)\s+(\d+(?:[.,]\d+)?)\s*(cm|centim[eè]tres?|m[eè]tres?|m)\b", lowered
    )
    if match:
        value = float(match.group(1).replace(",", "."))
        return value if match.group(2).startswith(("cm", "centim")) else value * 100.0
    distance = _extract_distance(text)
    if distance is None:
        return None
    return distance[0] if distance[1] == "cm" else distance[0] * 100.0


def _extract_target_object(text: str) -> str | None:
    """Extrait un nom d'objet simple depuis une phrase (heuristique légère)."""
    lowered = text.lower()
    patterns = [
        r"(?:objet|object)\s+(?:de type\s+)?([a-z0-9_\-éèêàùïîç]+)",
        r"(?:cherche|trouve|détecte|detecte|approche)\s+(?:un|une|le|la|l')?\s*([a-z0-9_\-éèêàùïîç]+)",
    ]
    forbidden = {
        "objet",
        "object",
        "toi",
        "point",
        "départ",
        "depart",
        "puis",
        "ensuite",
        "et",
        "de",
        "du",
        "la",
        "le",
        "les",
        "un",
        "une",
    }
    for pattern in patterns:
        match = re.search(pattern, lowered)
        if match:
            token = match.group(1).strip(" '-")
            if token and token not in forbidden:
                return token
    return None


class TextIntentParser:
    """Parse localement du texte libre en intentions métier structurées."""

    def parse(self, text: str) -> Intent:
        cleaned = (text or "").strip()
        if not cleaned:
            return Intent(name="unknown", raw_text=text)

        lowered = cleaned.lower()
        if any(phrase in lowered for phrase in ("suis-moi", "suis moi", "suivez-moi")):
            return Intent(name="follow_person", target="marker", confidence=1.0, raw_text=text)
        if re.fullmatch(
            r"(?:(?:approche|rapproche)(?: toi)? (?:de|vers) moi|rejoins moi)", fold(cleaned).strip(),
        ):
            return Intent(name="search_and_approach_object", target="marker", confidence=1.0, raw_text=text)

        refusal = follow_request_refusal(cleaned)
        if refusal:
            return Intent(name="unknown", raw_text=text)

        # "arrête-toi quand tu vois X" est une condition d'arrêt, pas un ordre d'arrêt immédiat.
        has_condition = any(
            token in lowered for token in ["quand", "jusqu", "dès que", "des que", "lorsque", "when", "until"]
        )
        has_search_verb = any(token in lowered for token in ["tourne", "cherche", "recherche", "scan", "look"])
        if has_condition and has_search_verb:
            color = "yellow" if any(t in lowered for t in ["jaune", "yellow"]) else (
                "red" if any(t in lowered for t in ["rouge", "red"]) else None
            )
            return Intent(
                name="search_and_approach_object",
                target=color,
                parameters={"search": True, "approach": False, **({"target_object": color} if color else {})},
                confidence=0.8,
                raw_text=text,
            )

        if any(token in lowered for token in ["stop", "arrêt", "arrete", "halt", "pause"]):
            return Intent(name="stop", confidence=0.95, raw_text=text)

        if any(token in lowered for token in ["retour au point de départ", "reviens au point de départ", "point de départ", "point de depart"]):
            return Intent(name="return_to_start", confidence=0.9, raw_text=text)

        if re.search(r"\b(?:suis|suit|suivre|suivez|follow)\b", lowered):
            color = _extract_color(lowered)
            parameters: dict[str, Any] = {}
            if color:
                parameters["target_object"] = color
            duration_s = _extract_duration(cleaned)
            if duration_s is not None:
                parameters["duration_s"] = duration_s
            stand_off_cm = _extract_stand_off_cm(cleaned)
            if stand_off_cm is not None:
                parameters["stand_off_cm"] = stand_off_cm
            return Intent(
                name="follow_object",
                target=color,
                parameters=parameters,
                confidence=0.88,
                raw_text=text,
            )

        if any(token in lowered for token in ["tour sur toi", "tourne sur toi", "rotation complète", "rotation complete", "360"]):
            return Intent(
                name="spin_360",
                parameters={"angle": 360, "unit": "degrees"},
                confidence=0.9,
                raw_text=text,
            )

        if any(token in lowered for token in ["carré", "carre", "forme un carré", "forme un carre"]):
            distance = _extract_distance(cleaned)
            side = distance[0] if distance else 40.0
            unit = distance[1] if distance else "cm"
            return Intent(
                name="move_square",
                parameters={"side": side, "unit": unit, "turn_direction": "right"},
                confidence=0.86,
                raw_text=text,
            )

        if any(token in lowered for token in ["contourne", "contourner", "évite l'objet", "evite l'objet", "évite objet", "evite objet"]):
            distance = _extract_distance(cleaned)
            parameters: dict[str, Any] = {"direction": "right", "clearance": 20.0, "unit": "cm"}
            if distance:
                parameters["clearance"] = distance[0]
                parameters["unit"] = distance[1]
            return Intent(
                name="contour_obstacle",
                parameters=parameters,
                confidence=0.84,
                raw_text=text,
            )

        if any(
            token in lowered
            for token in [
                "détecte", "detecte", "chercher un objet", "cherche un objet", "trouve un objet",
                "approche l'objet", "approcher l'objet", "s'en approcher", "s en approcher",
                "detect and approach", "find object", "approach object",
            ]
        ):
            target_object = _extract_target_object(cleaned)
            parameters: dict[str, Any] = {"search": True, "approach": True}
            if target_object:
                parameters["target_object"] = target_object
            return Intent(
                name="search_and_approach_object",
                target=target_object,
                parameters=parameters,
                confidence=0.87,
                raw_text=text,
            )

        color = _extract_color(lowered)
        has_approach_verb = re.search(r"\b(?:approche|approcher|rapproche|rapprocher)\b", lowered)
        has_search_verb_local = re.search(r"\b(?:cherche|chercher|recherche|trouve|look for|find)\b", lowered)
        if has_approach_verb or (color == "yellow" and has_search_verb_local):
            parameters = {"search": True, "approach": True}
            if color:
                parameters["target_object"] = color
            stand_off_cm = _extract_stand_off_cm(cleaned) if has_approach_verb else None
            if stand_off_cm is not None:
                parameters["stand_off_cm"] = stand_off_cm
            return Intent(
                name="search_and_approach_object",
                target=color,
                parameters=parameters,
                confidence=0.87,
                raw_text=text,
            )

        if any(token in lowered for token in ["balle rouge", "rouge", "cible rouge", "red ball"]):
            return Intent(
                name="follow_red_ball",
                target="red_ball",
                parameters={"speed": 0.5},
                confidence=0.9,
                raw_text=text,
            )

        if any(token in lowered for token in ["évite", "evite", "avoid", "obstacle"]):
            return Intent(
                name="avoid_obstacle",
                parameters={"speed": 0.4},
                confidence=0.85,
                raw_text=text,
            )

        if any(token in lowered for token in ["avance", "avancer", "go forward", "forward", "move forward", "va tout droit"]):
            return Intent(
                name="move_forward",
                parameters=_motion_parameters(cleaned),
                confidence=0.88,
                raw_text=text,
            )

        if any(token in lowered for token in ["recule", "reculer", "go back", "backward", "move backward", "arriere", "arrière"]):
            return Intent(
                name="move_backward",
                parameters=_motion_parameters(cleaned),
                confidence=0.88,
                raw_text=text,
            )

        if any(token in lowered for token in ["gauche", "turn left", "left turn", "tourne à gauche", "tourne a gauche"]):
            return Intent(
                name="turn_left",
                parameters=_turn_parameters(cleaned),
                confidence=0.88,
                raw_text=text,
            )

        if any(token in lowered for token in ["droite", "turn right", "right turn", "tourne à droite", "tourne a droite"]):
            return Intent(
                name="turn_right",
                parameters=_turn_parameters(cleaned),
                raw_text=text,
            )

        if any(token in lowered for token in ["cherche", "recherche", "look for", "find"]):
            return Intent(
                name="follow_red_ball",
                target="red_ball",
                confidence=0.7,
                raw_text=text,
            )

        return Intent(name="unknown", confidence=0.0, raw_text=text)

    _SEGMENT_SPLIT_RE = re.compile(r"\s*(?:(?<!\d),(?!\d)|\bet puis\b|\bpuis\b|\bensuite\b)\s*")
    _PRONOUN_RE = re.compile(r"\b(?:la|le|l'|lui)\b|-(?:la|le|lui)\b")

    def parse_plan(self, text: str) -> list[Intent]:
        """Découpe une phrase composée en segments et retourne la liste d'intents ordonnée."""
        if follow_request_refusal(text):
            return [self.parse(text)]
        segments = [seg for seg in self._SEGMENT_SPLIT_RE.split((text or "").strip()) if seg.strip()]
        if len(segments) <= 1:
            return [self.parse(text)]

        plan: list[Intent] = []
        last_target: str | None = None
        last_stand_off: float | None = None
        for segment in segments:
            intent = self.parse(segment)
            intent.raw_text = text
            lowered = segment.lower()

            # « cherche la balle rouge » suivi d'une approche/suivi : recherche, pas poursuite directe.
            if intent.name == "follow_red_ball" and re.search(r"\b(?:cherche|recherche)\b", lowered):
                intent = Intent(
                    name="search_and_approach_object",
                    target="red",
                    parameters={"search": True, "approach": True, "target_object": "red"},
                    confidence=intent.confidence,
                    raw_text=text,
                )

            if intent.name in {"search_and_approach_object", "follow_object"}:
                if intent.target is None and last_target and self._PRONOUN_RE.search(lowered):
                    intent.target = last_target
                    intent.parameters["target_object"] = last_target
                if intent.name == "follow_object" and "stand_off_cm" not in intent.parameters and last_stand_off:
                    intent.parameters["stand_off_cm"] = last_stand_off
                if intent.target in {"yellow", "red"}:
                    last_target = intent.target
                if "stand_off_cm" in intent.parameters:
                    last_stand_off = float(intent.parameters["stand_off_cm"])

            previous = plan[-1] if plan else None
            if (
                previous is not None
                and previous.name == intent.name == "search_and_approach_object"
                and (previous.target is None or intent.target is None or previous.target == intent.target)
            ):
                previous.parameters.update(intent.parameters)
                previous.target = intent.target or previous.target
                previous.confidence = min(previous.confidence, intent.confidence)
                continue
            plan.append(intent)

        return plan

"""Maps validated intents to skills and executes them."""

from __future__ import annotations

import logging

from irobot.domain.orchestrator.executor import SkillExecutor
from irobot.domain.orchestrator.planner import SkillPlanStep, SkillPlanner
from irobot.domain.orchestrator.registry import SkillRegistry
from irobot.domain.intent import Intent
from irobot.domain.skills.blackboard_state import BlackboardState
from irobot.domain.skills.base_skill import SkillContext
from irobot.domain.skills.behaviors.target_tracking import FOLLOW_INTENTS
from irobot.domain.intent_validator import IntentValidator
from irobot.interfaces.perception_contracts import ObjectPerceptionPipeline
from irobot.interfaces.plan_guards import follow_request_refusal


class LanguageController:
    """Executes planned intents through their associated skills."""

    def __init__(
        self,
        robot=None,
        camera=None,
        sensor=None,
        perception: ObjectPerceptionPipeline | None = None,
        registry: SkillRegistry | None = None,
    ):
        self.robot = robot
        self.camera = camera
        self.sensor = sensor
        self.perception = perception
        self.validator = IntentValidator()
        self.planner = SkillPlanner(registry=registry or SkillRegistry.default())

    def _inject_perception_state(self, context: SkillContext) -> None:
        if self.perception is None or context.perception_target is None:
            return
        detection = self.perception.detect(target=context.perception_target)
        if detection is None:
            return
        context.state["last_object_position"] = detection.position
        if detection.distance_cm is not None:
            context.state["last_object_distance_cm"] = float(detection.distance_cm)
        context.state["target_object"] = detection.label

    def _should_prioritize_obstacle_stop(self, context: SkillContext, skill_name: str) -> bool:
        if context.robot is None:
            return False
        check_obstacle = getattr(context.robot, "check_obstacle", None)
        if check_obstacle is None:
            return False
        if skill_name not in {"move_forward", "turn_left", "turn_right", "approach_target", "follow_object"}:
            return False
        try:
            return bool(check_obstacle())
        except Exception:
            logging.getLogger(__name__).exception("Echec du capteur d'obstacle")
            return True

    def handle_intent(self, intent: Intent):
        rejection = self._intent_rejection(intent)
        if rejection is not None:
            return rejection
        skills = self.planner.plan(intent.name)
        return self._execute_planned_step(SkillPlanStep(intent, tuple(skills)))

    def handle_planned_step(self, step: SkillPlanStep):
        rejection = self._intent_rejection(step.intent)
        if rejection is not None:
            return rejection
        return self._execute_planned_step(step)

    def _intent_rejection(self, intent: Intent):
        refusal = follow_request_refusal(intent.raw_text)
        if refusal:
            return {"intent": intent, "status": "rejected", "reason": refusal}
        if not self.validator.validate(intent):
            return {"intent": intent, "status": "rejected", "reason": "intent not allowed"}
        return None

    def _execute_planned_step(self, step: SkillPlanStep):
        intent = step.intent
        plan = step.skills
        if not plan:
            return {"intent": intent, "status": "rejected", "reason": "no skill plan"}

        context = SkillContext(
            robot=self.robot,
            camera=self.camera,
            sensor=self.sensor,
            perception=self.perception,
            perception_target=intent.perception_target,
            state=BlackboardState(
                active_intent=intent,
                active_intent_name=intent.name,
                active_intent_parameters={
                    **intent.parameters,
                    **({"target_object": intent.target} if intent.name in FOLLOW_INTENTS and intent.target else {}),
                },
                last_camera_position=None,
                last_object_position=None,
                last_ball_position=None,
                camera_ready=self.camera is not None,
            ),
        )

        executor = SkillExecutor(context)
        results = []
        for skill in plan:
            self._inject_perception_state(context)
            if self._should_prioritize_obstacle_stop(context, skill.name):
                stop_skill = self.planner.registry.get("stop_motion")
                if stop_skill is not None:
                    results.append(executor.run(stop_skill))
                return {
                    "intent": intent,
                    "status": "aborted",
                    "reason": "obstacle prioritized",
                    "plan": [skill.name for skill in plan],
                    "results": results,
                }
            context.state["active_skill"] = skill.name
            results.append(executor.run(skill))
        return {
            "intent": intent,
            "status": "accepted",
            "plan": [skill.name for skill in plan],
            "results": results,
        }

def fold(text: str) -> str:
    """Minuscules, sans accents ni ponctuation (comparaison lexicale)."""
    t = unicodedata.normalize("NFD", (text or "").lower())
    t = "".join(c for c in t if unicodedata.category(c) != "Mn")
    return re.sub(r"[^a-z0-9 ]+", " ", t)


def is_pure_stop(text: str) -> bool:
    """Phrase composée uniquement de mots d'arrêt : traitée localement, sans LLM."""
    words = fold(text).split()
    return bool(words) and any(w.startswith(("stop", "arret", "halte", "pause", "attend")) for w in words) \
        and all(w in STOP_WORDS or w.startswith(("arret", "attend")) for w in words)


def requests_human_follow(text: str) -> bool:
    """Identifies human-follow wording so target validation does not reject it."""
    normalized = fold(text)
    return bool(re.search(
        r"\b(?:suis|suit|suivre|suivez|suivez moi|follow|track|poursuis|rejoins)\s+"
        r"(?:(?!(?:balles?|balls?|objets?|objects?)\b)[a-z0-9]+\s+)*"
        r"(?:moi|nous|me|us|personne|humain|humaine|homme|femme|person|human|"
        r"porteur|utilisateur|operateur|proprietaire|quelqu un|mari|ami|amie|enfant|"
        r"wife|husband|child|children|friend)s?\b",
        normalized,
    )) or bool(re.search(r"\b(?:reprends|reprendre|reprenez)\s+(?:le\s+)?suivi\b", normalized))


TARGET_COLOR_SYNONYMS = {
    "jaune": "yellow", "yellow": "yellow", "balle jaune": "yellow", "yellow_ball": "yellow",
    "rouge": "red", "red": "red", "balle rouge": "red", "red_ball": "red",
    "marker": "marker",
}

FOLLOW_INTENTS = {
    "follow_object", "follow_red_ball", "follow", "track", "track_object",
    "follow_target", "follow_person", "suivre", "suit",
}


def supported_follow_target(value: Any) -> bool:
    return value is None or (
        isinstance(value, str) and value.strip().lower() in TARGET_COLOR_SYNONYMS
    )


def follow_request_refusal(text: str) -> str | None:
    for segment in re.split(r"\b(?:puis|ensuite|then|and|et)\b", fold(text)):
        if requests_human_follow(segment):
            continue
        follow = re.search(r"\b(?:suis|suit|suivre|suivez|follow|track|poursuis|rejoins)\b", segment)
        if follow is None:
            continue
        target = segment[follow.end():].strip()
        known = re.match(
            r"(?:(?:la|le|les|un|une|des|l|the|a|an)\s+)?"
            r"(?:(?:balles?|balls?|objets?|objects?)(?:\s+(?:jaune|rouge|yellow|red))?"
            r"|(?:jaune|rouge|yellow|red)(?:\s+(?:balles?|balls?|objets?|objects?))?)\b"
            r"|(?:la|le|les|l|it|them)\b",
            target,
        )
        if known is None:
            return "unsupported_follow_target"
        suffix = target[known.end():].strip()
        numeric_suffix = re.match(
            r"\d+(?:[.,]\d+)?\s*(?:secondes?|seconds?|secs?|s|minutes?|mins?|cm|metres?|m)\b",
            fold(fr_numbers_to_digits(suffix)),
        )
        if suffix and not numeric_suffix and not re.match(
            r"(?:pendant|pour|durant|for|during|a|at|de|du|des|d|of|jusqu|until|quand|when|please|svp)\b",
            suffix,
        ):
            return "unsupported_follow_target"
    return None


def contains_unsupported_follow_step(value: Any) -> bool:
    if isinstance(value, dict):
        names = {str(value.get(key, "")).strip().lower() for key in ("intent", "name", "action")}
        if names & FOLLOW_INTENTS:
            parameters = value.get("parameters")
            if parameters is not None and not isinstance(parameters, dict):
                return True
            if not supported_follow_target(value.get("target")):
                return True
            if isinstance(parameters, dict) and not supported_follow_target(parameters.get("target_object")):
                return True
        return any(contains_unsupported_follow_step(item) for item in value.values())
    if isinstance(value, list):
        return any(contains_unsupported_follow_step(item) for item in value)
    return False


def _num(value: Any) -> float | None:
    if isinstance(value, bool):
        return None
    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def _licensed(intent: str, folded: str) -> bool:
    rule = INTENT_LEXICON.get(intent)
    if rule is None:          # intent sans règle (execute_path...) : pas de contrôle
        return True
    prefixes, words, phrases = rule
    tokens = folded.split()
    padded = f" {' '.join(tokens)} "
    return (any(t.startswith(p) for t in tokens for p in prefixes)
            or any(t in words for t in tokens)
            or any(f" {ph} " in padded for ph in phrases))


def marker_target_mismatch(text: str, step: dict[str, Any]) -> bool:
    folded = fold(text)
    params = step.get("parameters")
    return (
        (step.get("intent") or step.get("name")) in {"search_and_approach_object", "follow_person"}
        and re.search(r"\bmoi\b", folded) is not None
        and re.search(r"\b(?:jaune|yellow|rouge|red)\b", folded) is None
        and (
            step.get("target") != "marker"
            or (isinstance(params, dict) and params.get("target_object", "marker") != "marker")
        )
    )


def grounding_check(sent_text: str, steps: list) -> tuple[list, str | None, list[str]]:
    """Retourne (étapes_conservées, raison_de_rejet | None, notes).

    - intent non justifié par un mot de la phrase  -> plan REJETÉ en entier ;
    - paramètre que l'intent ne gère pas          -> supprimé (bruit du LLM) ;
    - `stop` non justifié                         -> étape supprimée (action sûre, mais parasite) ;
    - valeur numérique absente de la phrase :
        * la phrase ne contient AUCUN chiffre -> paramètre inventé, supprimé (défaut du runtime) ;
        * la phrase contient des chiffres     -> le LLM les a mal repris : plan REJETÉ ;
      (valeurs par défaut du prompt tolérées : duration 1 pour avance/recule, angle 90 pour tourne) ;
    - après nettoyage, avance/recule sans distance ni durée -> duration 1 ; tourne sans angle ni durée -> angle 90.
    """
    refusal = follow_request_refusal(sent_text)
    if refusal:
        return [], refusal, []
    folded = fold(sent_text)
    numbers = {float(x.replace(",", ".")) for x in re.findall(r"\d+(?:[.,]\d+)?", folded)}
    kept, notes = [], []
    for step in steps:
        if not isinstance(step, dict):
            return [], "étape invalide", notes
        intent = step.get("intent") or step.get("name")
        if marker_target_mismatch(sent_text, step):
            return [], "moi désigne le marqueur : target doit être marker", notes
        if not _licensed(intent, folded):
            if intent == "stop":
                notes.append("stop parasite supprimé")
                continue
            return [], f"intent {intent!r} non justifié par la phrase", notes
        params = step.get("parameters")
        if isinstance(params, dict):
            cleaned = {}
            allowed = ALLOWED_PARAMS.get(intent, set())
            for key, value in params.items():
                if intent in INTENT_LEXICON and key not in allowed:
                    notes.append(f"{intent}.{key} inutile : supprimé")
                    continue
                num = _num(value) if key in NUMERIC_KEYS else None
                if key in NUMERIC_KEYS and num is not None and num not in numbers:
                    default = ((key == "duration" and intent in {"move_forward", "move_backward"} and num == 1)
                               or (key == "angle" and intent in {"turn_left", "turn_right"} and num == 90))
                    if not default:
                        if numbers:
                            return [], f"{intent}.{key}={value} ne correspond à aucun nombre de la phrase", notes
                        notes.append(f"{intent}.{key}={value} absent de la phrase : supprimé")
                        continue
                cleaned[key] = value
            step = {**step, "parameters": cleaned}
        if intent in {"move_forward", "move_backward"}:
            p = dict(step.get("parameters") or {})
            if not any(k in p for k in ("distance", "duration")):
                p["duration"] = 1
                step = {**step, "parameters": p}
        elif intent in {"turn_left", "turn_right"}:
            p = dict(step.get("parameters") or {})
            if not any(k in p for k in ("angle", "duration")):
                p["angle"] = 90
                step = {**step, "parameters": p}
        kept.append(step)
    return kept, None, notes

from __future__ import annotations

import logging
import math
import threading
import time
from collections.abc import Callable
from typing import Any, Protocol


class DistanceSensor(Protocol):
    def read_distance_cm(self) -> float | None: ...


class RobotController:
    """Adapter between robot business logic and hardware implementation."""

    CONTROL_LOOP_PERIOD_S = 0.02
    SENSOR_READ_ATTEMPTS = 5
    SENSOR_RETRY_DELAY_S = 0.03

    def __init__(
        self,
        robot: Any,
        time_fn: Callable[[], float] = time.monotonic,
        sleep_fn: Callable[[float], None] = time.sleep,
        *,
        distance_sensor: DistanceSensor | None = None,
        obstacle_threshold_cm: float = 5.0,
        require_obstacle_sensor: bool = False,
    ):
        self._robot = robot
        self.time_fn = time_fn
        self.sleep_fn = sleep_fn
        self._distance_sensor = distance_sensor
        self._obstacle_threshold_cm = obstacle_threshold_cm
        self._require_obstacle_sensor = bool(require_obstacle_sensor)
        self._emergency_event = threading.Event()

    @property
    def robot(self):
        return self._robot

    @property
    def speed(self):
        return self._robot.speed

    def _start_motion(self, method: str) -> None:
        if self._emergency_event.is_set():
            self.stop()
            return
        getattr(self._robot, method)()
        # A stop may arrive while the driver is starting the motion.
        if self._emergency_event.is_set():
            self.stop()

    def forward(self) -> None:
        self._start_motion("forward")

    def backward(self) -> None:
        self._start_motion("backward")

    def left(self) -> None:
        self._start_motion("left")

    def right(self) -> None:
        self._start_motion("right")

    def stop(self) -> None:
        """Deliver zero immediately, without waiting for sensors or motion loops."""
        self._robot.stop()

    def set_speed(self, speed: float) -> None:
        if speed != 0 and self._emergency_event.is_set():
            self.stop()
            return
        self._robot.set_speed(speed)
        if self._emergency_event.is_set():
            self.stop()

    def check_obstacle(self) -> bool:
        if self._distance_sensor is not None:
            return self._check_distance_sensor()
        if self._require_obstacle_sensor:
            return True

        check = getattr(self._robot, "check_obstacle", None)
        if not callable(check):
            return False
        try:
            obstacle = check()
        except Exception:
            logging.getLogger(__name__).exception("Echec du capteur d'obstacle")
            return True
        if obstacle is None:
            logging.getLogger(__name__).error("Capteur d'obstacle indisponible")
            return True
        return bool(obstacle)

    def _check_distance_sensor(self) -> bool:
        sensor = self._distance_sensor
        if sensor is None:
            return self._require_obstacle_sensor

        for attempt in range(self.SENSOR_READ_ATTEMPTS):
            try:
                distance_cm = sensor.read_distance_cm()
            except Exception:
                logging.getLogger(__name__).exception("Echec du capteur d'obstacle")
                return True
            if distance_cm is not None:
                return distance_cm < self._obstacle_threshold_cm
            if attempt < self.SENSOR_READ_ATTEMPTS - 1:
                self.sleep_fn(self.SENSOR_RETRY_DELAY_S)
        logging.getLogger(__name__).error("Capteur d'obstacle indisponible")
        return True

    def request_emergency_stop(self) -> None:
        self._emergency_event.set()
        self.stop()

    def clear_emergency_stop(self) -> None:
        self._emergency_event.clear()

    def is_emergency_stop_requested(self) -> bool:
        return self._is_emergency_stop_requested()

    def _is_emergency_stop_requested(self) -> bool:
        if self._emergency_event.is_set():
            return True
        for name in (
            "is_emergency_stop_requested", "check_emergency_stop",
            "emergency_stop_requested", "stop_requested",
        ):
            detector = getattr(self._robot, name, None)
            if detector is None:
                continue
            try:
                value = detector() if callable(detector) else detector
            except Exception:
                logging.getLogger(__name__).exception("Echec de l'entree d'arret %s", name)
                return True
            if value:
                return True
        return False

    def _run_motion_for(
        self, start_motion: Callable[[], None], duration_seconds: float,
        check_obstacle: bool = True,
    ) -> str | None:
        try:
            if (isinstance(duration_seconds, bool)
                    or not isinstance(duration_seconds, (int, float))
                    or not math.isfinite(duration_seconds) or duration_seconds < 0):
                raise ValueError("Motion duration must be finite and non-negative")
            if self._is_emergency_stop_requested():
                return "emergency_stop"
            if check_obstacle and self.check_obstacle():
                return "obstacle"
            if self._is_emergency_stop_requested():
                return "emergency_stop"
            start_motion()
            if duration_seconds == 0:
                return None
            deadline = self.time_fn() + duration_seconds
            while True:
                if self._is_emergency_stop_requested():
                    return "emergency_stop"
                if check_obstacle and self.check_obstacle():
                    return "obstacle"
                if self._is_emergency_stop_requested():
                    return "emergency_stop"
                remaining = deadline - self.time_fn()
                if remaining <= 0.0:
                    return None
                self.sleep_fn(min(self.CONTROL_LOOP_PERIOD_S, remaining))
        finally:
            self.stop()

    def forward_for(self, duration_seconds: float) -> str | None:
        return self._run_motion_for(self.forward, duration_seconds)

    def backward_for(self, duration_seconds: float) -> str | None:
        return self._run_motion_for(self.backward, duration_seconds, check_obstacle=False)

    def left_for(self, duration_seconds: float) -> str | None:
        return self._run_motion_for(self.left, duration_seconds)

    def right_for(self, duration_seconds: float) -> str | None:
        return self._run_motion_for(self.right, duration_seconds)

    def __getattr__(self, name: str):
        return getattr(self._robot, name)

"""Prétraitement déterministe des transcriptions vocales avant interprétation.

Porté depuis `scripts/scenarios/eval_prompt_plan.py` ; `preprocess` gère en plus les durées et
distances composées (« 1 minute 30 secondes », « 1 mètre 50 »). Un petit LLM convertit mal les
nombres en lettres, on lui envoie donc un texte déjà normalisé (nombres en chiffres,
minutes -> secondes, mètres -> centimètres).
"""

from __future__ import annotations

import re

_UNITS = {
    "zéro": 0, "zero": 0, "un": 1, "une": 1, "deux": 2, "trois": 3, "quatre": 4, "cinq": 5,
    "six": 6, "sept": 7, "huit": 8, "neuf": 9, "dix": 10, "onze": 11, "douze": 12, "treize": 13,
    "quatorze": 14, "quinze": 15, "seize": 16, "dix-sept": 17, "dix-huit": 18, "dix-neuf": 19,
}
_TENS = {"vingt": 20, "vingts": 20, "trente": 30, "quarante": 40, "cinquante": 50, "soixante": 60}
_UNIT_WORDS = {
    "mètre", "mètres", "metre", "metres", "centimètre", "centimètres", "centimetre", "centimetres",
    "cm", "m", "seconde", "secondes", "minute", "minutes", "degré", "degrés", "degre", "degres",
}


def _tokens(text: str) -> list[str]:
    text = text.lower().replace("’", "'")
    text = re.sub(r"\b([dlcjnmst])'(?=\w)", r"\1' ", text)  # d'un -> d' un
    return [t for t in re.split(r"[\s\-,;.!?]+", text) if t]


def fr_numbers_to_digits(text: str) -> str:
    """Remplace les nombres français en lettres par des chiffres (« un/une » seulement devant une unité)."""
    toks = _tokens(text)
    out: list[str] = []
    i = 0
    while i < len(toks):
        tok = toks[i]
        starts = tok in _UNITS or tok in _TENS or tok in ("cent", "cents", "mille")
        if not starts:
            out.append(tok)
            i += 1
            continue
        if tok in ("un", "une") and not (i + 1 < len(toks) and toks[i + 1] in _UNIT_WORDS):
            out.append(tok)
            i += 1
            continue
        total = cur = 0
        j, prev, consumed = i, None, 0
        while j < len(toks):
            t = toks[j]
            if t == "et" and consumed and j + 1 < len(toks) and (toks[j + 1] in _UNITS or toks[j + 1] in _TENS):
                j += 1
                continue
            if t in _UNITS:
                v = _UNITS[t]
                m = cur % 100
                ok = (
                    (m == 0 and not (consumed and prev in _UNITS))
                    or (m in (20, 30, 40, 50, 60, 70, 80, 90) and v < 10)
                    or (m in (60, 80) and 10 <= v < 20)
                    or (m == 10 and 7 <= v <= 9)
                )
                if t in ("un", "une") and consumed == 0:
                    ok = True
                if not ok:
                    break
                cur += v
            elif t in _TENS:
                if t in ("vingt", "vingts") and prev == "quatre" and cur % 10 == 4:
                    cur = cur - 4 + 80
                elif cur % 100 == 0:
                    cur += _TENS[t]
                else:
                    break
            elif t in ("cent", "cents"):
                cur = (cur or 1) * 100 if cur < 100 else cur
            elif t == "mille":
                total += (cur or 1) * 1000
                cur = 0
            else:
                break
            prev = t
            consumed += 1
            j += 1
        out.append(str(total + cur))
        i = j
    return " ".join(out)


_COMPOUND_DURATION_RE = re.compile(r"\b(\d+)\s+minutes?\s+(?:et\s+)?(\d+)\s+secondes?\b")
# « N mètre(s) [et] M [centimètre(s)] », M sur 1 ou 2 chiffres ; sans « centimètres », M ne doit pas
# être suivi d'une autre unité (« 1 mètre et 2 secondes » n'est pas une distance composée).
_COMPOUND_DISTANCE_RE = re.compile(
    r"\b(\d+)\s+m[eè]tres?\s+(?:et\s+)?(\d{1,2})"
    r"(?:\s+(?:centim[eè]tres?|cm)\b"
    r"|(?!\d|\s+(?:secondes?|minutes?|degr[eé]s?|m[eè]tres?|m|fois|tours?)\b))"
)


def preprocess(text: str) -> str:
    """Prétraitement déterministe avant le LLM : nombres en chiffres, minutes -> secondes, mètres -> cm.

    Les formes composées (« 1 minute 30 secondes », « 1 mètre 50 ») sont converties avant les formes simples.
    """
    t = fr_numbers_to_digits(text)
    t = _COMPOUND_DURATION_RE.sub(lambda m: f"{int(m.group(1)) * 60 + int(m.group(2))} secondes", t)
    t = _COMPOUND_DISTANCE_RE.sub(lambda m: f"{int(m.group(1)) * 100 + int(m.group(2))} centimètres", t)
    t = re.sub(r"\b(\d+)\s+minutes?\b", lambda m: f"{int(m.group(1)) * 60} secondes", t)
    t = re.sub(r"\b(\d+)\s+m[eè]tres?\b", lambda m: f"{int(m.group(1)) * 100} centimètres", t)
    return t

def _parse_llm_json_payload(content: str) -> dict[str, Any] | None:
    """Extrait un objet JSON depuis une réponse LLM, y compris si elle est entourée de backticks."""
    cleaned = content.strip()
    if not cleaned:
        return None
    if cleaned.startswith("```"):
        cleaned = cleaned.strip("`")
        if cleaned.lower().startswith("json"):
            cleaned = cleaned.split("\n", 1)[1].strip() if "\n" in cleaned else cleaned[4:].strip()
    try:
        payload = json.loads(cleaned)
        if isinstance(payload, dict):
            return payload
    except json.JSONDecodeError:
        pass
    return None


def _llm_payload(system_prompt: str, user_text: str, max_tokens: int, model: str) -> dict[str, Any]:
    return {
        "model": model,
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_text},
        ],
        "temperature": 0.0,
        "max_tokens": max_tokens,
        "stream": False,
        "cache_prompt": True,
    }


class ResponseFormatChain:
    """`response_format` essayés dans l'ordre (None = sans) ; le premier accepté est mémorisé."""

    def __init__(self, formats: list[dict[str, Any] | None]):
        self.formats = list(formats)
        self._start = 0

    def candidates(self) -> list[tuple[int, dict[str, Any] | None]]:
        return list(enumerate(self.formats))[self._start:]

    def accept(self, index: int) -> None:
        self._start = index

    def reset(self) -> None:
        self._start = 0

    ) -> str: ...

def llm_response_format_mode(environ: Any = None) -> str:
    """Lit LLM_RESPONSE_FORMAT (schema | json_object | none) ; valeur absente ou inconnue -> schema."""
    env = os.environ if environ is None else environ
    mode = (env.get("LLM_RESPONSE_FORMAT") or "").strip().lower()
    return mode if mode in LLM_RESPONSE_FORMAT_MODES else "schema"


def _select_formats(schema_chain: ResponseFormatChain) -> ResponseFormatChain:
    mode = llm_response_format_mode()
    if mode == "json_object":
        return ResponseFormatChain([{"type": "json_object"}])
    if mode == "none":
        return ResponseFormatChain([None])
    return schema_chain


def _response_format_kind(fmt: dict[str, Any] | None) -> str:
    """json_schema | json_object_schema | json_object | none."""
    if fmt is None:
        return "none"
    kind = str(fmt.get("type"))
    if kind == "json_object" and "schema" in fmt:
        return "json_object_schema"
    return kind


def llm_warmup_wait_seconds(environ: Any = None) -> float:
    """Lit LLM_WARMUP_WAIT_SECONDS (>= 0) ; valeur absente ou invalide -> 60 s."""
    env = os.environ if environ is None else environ
    try:
        value = float(env["LLM_WARMUP_WAIT_SECONDS"])
    except (KeyError, TypeError, ValueError):
        return LLM_WARMUP_WAIT_SECONDS
    return value if math.isfinite(value) and value >= 0 else LLM_WARMUP_WAIT_SECONDS


def _post_llm_chat(
    payload: dict[str, Any],
    timeout: tuple[float, float],
    formats: ResponseFormatChain,
    meta: dict[str, Any] | None = None,
) -> str:
    """Appel non streamé ; format suivant sur HTTP 400 ; retourne le contenu texte de la réponse.

    `meta` (optionnel) reçoit la durée réelle, le nombre de requêtes, le format utilisé et
    `usage`/`timings` de llama-server, y compris quand l'appel échoue.
    """
    meta = {} if meta is None else meta
    meta.update(
        llm_latency_seconds=None,
        llm_response_format=None,
        llm_attempts=0,
        llm_prompt_tokens=None,
        llm_completion_tokens=None,
        llm_timings=None,
    )
    started = time.monotonic()
    try:
        candidates = formats.candidates()
        response = None
        index = 0
        for position, (index, fmt) in enumerate(candidates):
            body = dict(payload)
            if fmt is not None:
                body["response_format"] = fmt
            meta["llm_attempts"] += 1
            meta["llm_response_format"] = _response_format_kind(fmt)
            response = requests.post(LLM_URL, json=body, timeout=timeout)
            if getattr(response, "status_code", 200) == 400 and position < len(candidates) - 1:
                continue
            break
        response.raise_for_status()
        formats.accept(index)
        data = response.json()
        usage = data.get("usage") if isinstance(data, dict) else None
        if isinstance(usage, dict):
            meta["llm_prompt_tokens"] = usage.get("prompt_tokens")
            meta["llm_completion_tokens"] = usage.get("completion_tokens")
        timings = data.get("timings") if isinstance(data, dict) else None
        if isinstance(timings, dict):
            meta["llm_timings"] = dict(timings)
        return str(data["choices"][0]["message"]["content"]).strip()
    finally:
        meta["llm_latency_seconds"] = round(time.monotonic() - started, 3)


def _llm_error_reason(exc: Exception, read_timeout: float) -> str:
    if isinstance(exc, requests.exceptions.ConnectionError):
        return "serveur LLM injoignable"
    if isinstance(exc, requests.exceptions.Timeout):
        return f"LLM trop lent (> {read_timeout:g}s)"
    if isinstance(exc, requests.exceptions.HTTPError):
        return f"erreur HTTP du serveur LLM ({exc})"
    return f"erreur LLM ({type(exc).__name__}: {exc})"

class VoiceRuntime:
    """Microphone/text adapter with immediate stop handling."""

    def __init__(
        self,
        *,
        robot=None,
        camera=None,
        sensor=None,
        perception: ObjectPerceptionPipeline | None = None,
        registry: SkillRegistry | None = None,
        input_device: str = "auto",
        use_llm: bool = True,
        trace_dir: str | None = None,
        threaded: bool = True,
    ):
        self.robot = robot
        self.input_device = input_device
        self.threaded = threaded
        self._worker: threading.Thread | None = None
        self.last_result: Any = None
        self.camera = camera
        self.perception = perception
        self.command_service = CommandService(
            robot=robot,
            camera=camera,
            sensor=sensor,
            perception=perception,
            registry=registry,
            use_llm=use_llm,
            trace_dir=trace_dir,
            llm_parse_with_trace=ask_llm_for_intent_plan_with_trace,
        )
        self._emergency_event = self.command_service._emergency_event

    def _safe_stop(self) -> None:
        self.command_service.safe_stop()

    def handle_text_as_plan(self, text: str):
        return self.command_service.handle_text(text)

    def listen_once(self):
        device = resolve_input_device(self.input_device)
        for text in transcribe_loop(device):
            print(f"\n[VOIX] {text}", flush=True)
            self._clear_emergency_stop()
            result = self.handle_text_as_plan(text)
            print(f"[RUNTIME] {result}", flush=True)
            return result
        return {"status": "no_transcript"}

    def _call_robot(self, method: str) -> None:
        func = getattr(self.robot, method, None)
        if callable(func):
            try:
                func()
            except Exception:
                logging.getLogger(__name__).exception("Echec de la commande robot %s", method)

    def _clear_emergency_stop(self) -> None:
        self._emergency_event.clear()
        self._call_robot("clear_emergency_stop")

    def _request_emergency_stop(self) -> None:
        self._emergency_event.set()
        if callable(getattr(self.robot, "request_emergency_stop", None)):
            self._call_robot("request_emergency_stop")
        else:
            self._safe_stop()

    def mission_running(self) -> bool:
        return self._worker is not None and self._worker.is_alive()

    def _is_stop_phrase(self, text: str) -> bool:
        return is_pure_stop(text) or re.search(
            r"\b(?:stop|arrete|arret|halte|halt|pause)\b", fold(text),
        ) is not None

    def _mission_worker(self, text: str) -> None:
        try:
            self.last_result = self.handle_text_as_plan(text)
            print(f"[RUNTIME] {self.last_result}", flush=True)
        except Exception as exc:
            self.last_result = {"status": "failed", "error": str(exc)}
            print(f"[RUNTIME] erreur mission : {exc}", flush=True)
            self._safe_stop()

    def handle_transcript(self, text: str):
        """Stop immédiat pendant une mission, sinon démarre une nouvelle mission."""
        if self._is_stop_phrase(text):
            try:
                if self.robot is not None:
                    self.robot.stop()
            finally:
                self._request_emergency_stop()
                self.command_service.mission_orchestrator.state_machine.abort("emergency stop")
            print("[RUNTIME] Arrêt d'urgence demandé.", flush=True)
            return None
        if self.mission_running():
            print("[RUNTIME] Mission en cours : phrase ignorée (dis « stop » pour arrêter).", flush=True)
            return None

        # Effacé ici (thread principal) et non dans le worker pour ne pas écraser un stop tardif.
        self._clear_emergency_stop()
        if not self.threaded:
            self.last_result = self.handle_text_as_plan(text)
            print(f"[RUNTIME] {self.last_result}", flush=True)
            return self.last_result
        self._worker = threading.Thread(
            target=self._mission_worker,
            args=(text,),
            name="mission-worker", daemon=True,
        )
        self._worker.start()
        return None

    def listen_forever(self):
        device = resolve_input_device(self.input_device)
        try:
            for text in transcribe_loop(device):
                print(f"\n[VOIX] {text}", flush=True)
                self.handle_transcript(text)
        except KeyboardInterrupt:
            print("\nInterruption clavier : arrêt du runtime vocal.")
            self._abort_mission()

    def _abort_mission(self) -> None:
        self._request_emergency_stop()
        worker = self._worker
        if worker is not None and worker.is_alive() and worker is not threading.current_thread():
            worker.join(timeout=2.0)
        self._safe_stop()

    def close(self) -> None:
        """Arrête proprement les périphériques matériels et les threads d'arrière-plan"""
        try:
            self._abort_mission()
        except Exception:
            logging.getLogger(__name__).exception("Impossible d'interrompre la mission avant la fermeture")
            try:
                close_hardware_runtime({"robot": self.robot, "camera": self.perception})
            except Exception:
                logging.getLogger(__name__).exception("Échec de la fermeture des ressources après interruption")
            raise
        close_hardware_runtime({"robot": self.robot, "camera": self.perception})

    parser = argparse.ArgumentParser(
        description="Runtime vocal complet : Vosk → intent → skills -> robot."
    )
    parser.add_argument(

        action="store_true",
        help="Ecoute une seule phrase puis s'arrête.",
    )
    parser.add_argument(
        "--no-llm",
        action="store_true",
        help="Utilise le parseur local au lieu du LLM pour interpréter la commande.",
    )
    parser.add_argument(
        "--input-device",
        default="auto",
        choices=["auto", "bluetooth", "wm8960", "default"],
        help="Source audio utilisée pour l'enregistrement.",
    )
    parser.add_argument(

        default=None,
        help="Traite cette phrase (sans micro) puis s'arrête.",
    )
    return parser.parse_args()


def main() -> None:
    args = _parse_args()
    runtime = None
    hardware = None
    try:
        warmup = LLMWarmup().start() if not args.no_llm else None
        hardware = build_hardware_runtime()
        runtime = VoiceRuntime(
            robot=hardware["robot"],
            camera=hardware["camera"],
            sensor=hardware["sensor"],
            perception=hardware["camera"],
            input_device=args.input_device,
            use_llm=not args.no_llm,
        )
        runtime.command_service.warmup = warmup
        # Une seule commande : la vraie requête attend le préchauffage ; écoute continue : arrière-plan.

            print(f"[RUNTIME] {result}", flush=True)
            return

    except RuntimeError as exc:

        sys.exit(1)
    finally:
        if runtime is not None:
            runtime.close()
        elif hardware is not None:
            close_hardware_runtime(hardware)

"""Outil de calibration caméra : affiche diameter_px et la distance estimée.

Usage :
    uv run calibrate-distance --color yellow

from __future__ import annotations

import argparse
import statistics
import sys

import cv2

from irobot.config.settings import load_runtime_settings

from irobot.hardware.camera.frame_enhancer import FrameEnhancer, configure_exposure

try:
    from picamera2 import Picamera2
except ModuleNotFoundError:  # pragma: no cover - environnements hors Raspberry Pi
    Picamera2 = None

CAMERA_INDEX = 0


def focal_from_known_distance(diameter_px: float, ball_diameter_cm: float, known_distance_cm: float) -> float | None:
    """Inverse du modèle : focal_px = diameter_px * known_distance_cm / ball_diameter_cm."""
    if diameter_px is None or diameter_px <= 0 or ball_diameter_cm <= 0 or known_distance_cm <= 0:
        return None
    return diameter_px * known_distance_cm / ball_diameter_cm


def parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    settings = load_runtime_settings()
    parser = argparse.ArgumentParser(description="Calibre l'estimation de distance de la balle.")
    parser.add_argument("--color", choices=["red", "yellow"], default="yellow")
    parser.add_argument("--ball-diameter-cm", type=float, default=settings.camera_ball_diameter_cm)
    parser.add_argument("--focal-px", type=float, default=settings.camera_focal_px)
    parser.add_argument("--known-distance-cm", type=float, default=None, help="distance réelle balle-caméra")
    parser.add_argument("--samples", type=int, default=30, help="mesures moyennées en mode --known-distance-cm")
    return parser.parse_args(argv)


def main(argv: list[str] | None = None) -> None:
    args = parse_args(argv)
    if Picamera2 is None:
        sys.exit("Picamera2 indisponible : à lancer sur le Raspberry Pi.")

    settings = load_runtime_settings()
    enhancer = FrameEnhancer(settings)
    detector = ColorDetector(args.color, args.ball_diameter_cm, args.focal_px)
    camera = Picamera2(CAMERA_INDEX)
    started = False
    samples: list[float] = []
    try:
        # Picamera2 BGR888 fournit des octets R, G, B au détecteur is_rgb=True.
        camera.configure(camera.create_preview_configuration(
            main={"size": settings.camera_capture_size_px, "format": "BGR888"},
        ))
        configure_exposure(camera, settings)
        camera.start()
        started = True
        while True:
            frame = cv2.rotate(camera.capture_array(), cv2.ROTATE_90_CLOCKWISE)
            frame = enhancer.apply(frame)
            detection = detector.detect(frame)
            if detection is None or not detection.label.endswith("_ball"):
                print("balle non détectée          ", end="\r", flush=True)
                continue
            diameter = detection.diameter_px
            if diameter is None:
                print("diamètre de balle indisponible          ", end="\r", flush=True)
                continue

            if args.known_distance_cm is None:
                distance = detection.distance_cm
                if distance is None:
                    print("distance indisponible          ", end="\r", flush=True)
                    continue
                print(f"diameter_px={diameter:7.1f}  distance={distance:7.1f} cm", end="\r", flush=True)
                continue

            samples.append(diameter)
            if len(samples) >= max(1, args.samples):
                median = statistics.median(samples)
                focal = focal_from_known_distance(median, args.ball_diameter_cm, args.known_distance_cm)
                print(f"\ndiameter_px (médiane) = {median:.1f}")
                print(f"À reporter dans [camera] : focal_px = {focal:.1f}")
                return
    except KeyboardInterrupt:
        print()
    finally:
        try:
            if started:
                camera.stop()
        finally:
            camera.close()


if __name__ == "__main__":
    main()

        2,
    )


def main(argv: list[str] | None = None) -> int:
    import argparse
    import os
    import sys

    parser = argparse.ArgumentParser(description="Diagnostic vision en direct.")
    parser.add_argument(
        "--preview-only",
        action="store_true",
        help="affiche le flux caméra sans exécuter les détecteurs ni ajouter d'annotations",
    )
    args = parser.parse_args(argv)

    if not os.environ.get("DISPLAY"):
        print("Erreur : aucun affichage X11 détecté (variable DISPLAY absente).", file=sys.stderr)
        return 2

    import cv2
    from picamera2 import Picamera2

    camera_info = Picamera2.global_camera_info()
    if not camera_info:
        print("Erreur : aucune caméra détectée.", file=sys.stderr)
        return 2
    model = str(camera_info[0].get("Model", "")).lower()
    if model != EXPECTED_CAMERA_MODEL:
        print(
            f"Erreur : caméra 0 = {model or 'modèle inconnu'}, "
            f"OV5647 attendu ; aucune caméra ouverte.",
            file=sys.stderr,
        )
        return 2

    settings = load_runtime_settings()
    enhancer = FrameEnhancer(settings)
    if not args.preview_only:
        marker_detector = OpenCVMarkerDetector(
            TEST_MARKER_IDENTITY,
            focal_px=settings.camera_focal_px,
            oriented_frame_size_px=settings.camera_oriented_frame_size_px,
        )
        faces = FaceDetector(settings)
    camera = Picamera2(0)
    started = False
    last_status: tuple | None = None
    if args.preview_only:
        print("Aperçu caméra uniquement. Aucune détection ni annotation. Appuyez sur q pour quitter.", flush=True)
    else:
        print(
            "Diagnostic vision : ArUco DICT_4X4_50 / ID 1, visages, balles rouge/jaune. "
            f"Distance directe estimée par pose (côté {MARKER_SIDE_CM:g} cm, "
            f"focale configurée {settings.camera_focal_px:g} px) ; "
            "objectif non calibré, distorsion non corrigée. "
            "Distances visage/balle indicatives. Visage mémorisé signalé pendant la grâce. "
            "Aucune image n'est enregistrée. Appuyez sur q pour quitter.",
            flush=True,
        )
    try:
        camera.configure(camera.create_preview_configuration(
            main={"size": settings.camera_capture_size_px, "format": "BGR888"},
        ))
        configure_exposure(camera, settings)
        camera.start()
        started = True
        while True:
            frame_rgb = camera.capture_array()
            rotated_rgb = cv2.rotate(frame_rgb, cv2.ROTATE_90_CLOCKWISE)
            rotated_rgb = enhancer.apply(rotated_rgb)
            display_frame = cv2.cvtColor(rotated_rgb, cv2.COLOR_RGB2BGR)
            if args.preview_only:
                cv2.imshow("Vision - Preview", display_frame)
            else:
                detection = marker_detector.detect(rotated_rgb)
                face = faces.detect(rotated_rgb)
                balls = detect_balls(rotated_rgb, settings)
                _draw_detection(display_frame, detection, cv2)
                statuses = [format_object_status("FACE", face, cached=faces.is_cached)]
                _draw_object_status(
                    display_frame, "FACE", face, 1, (255, 200, 0), cv2, cached=faces.is_cached,
                )
                if faces.bounding_box is not None:
                    x, y, width, height = map(round, faces.bounding_box)
                    cv2.rectangle(display_frame, (x, y), (x + width, y + height), (255, 200, 0), 2)
                for row, (color, result) in enumerate(balls.items(), start=2):
                    ball = result[0] if result else None
                    text = format_object_status(f"BALL {color.upper()}", ball)
                    statuses.append(text)
                    box_color = (0, 0, 255) if color == "red" else (0, 255, 255)
                    _draw_object_status(display_frame, f"BALL {color.upper()}", ball, row, box_color, cv2)
                    if result:
                        x, y, width, height = result[1]
                        cv2.rectangle(display_frame, (x, y), (x + width, y + height), box_color, 2)
                cv2.imshow("Vision - ArUco / Face / Red / Yellow", display_frame)

                status_key = (detection.disposition, detection.reason, *statuses)
                if status_key != last_status:
                    print(format_detection_status(detection), flush=True)
                    print(" | ".join(statuses), flush=True)
                    last_status = status_key

            if cv2.waitKey(1) & 0xFF == ord("q"):
                break
    except KeyboardInterrupt:
        print("\nDiagnostic interrompu par l'utilisateur.", flush=True)
        return 130
    finally:
        try:
            if started:
                camera.stop()
        finally:
            camera.close()
            cv2.destroyAllWindows()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
